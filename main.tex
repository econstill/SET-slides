\begin{frame}{Equilibrium uniqueness and representation}
In nonlinear models, standard sufficient conditions for uniqueness
can be difficult to verify.

Some of this difficulty may come from how the equilibrium
conditions are written.

\begin{block}{The central question}
How can we transform nonlinear equilibrium to simplify analysis, without changing the equilibrium set?
\end{block}

\medskip
The paper develops \alert{sign-equivalent transformations (SET)}
to reveal useful mathematical structure in equilibrium problems.
\end{frame}


\begin{frame}{The paper's map}
\begin{center}
\includegraphics[width=0.87\textwidth]{assets/set-figure1-methodology.png}
\end{center}
\vspace{-0.25cm}
\small An equilibrium is a best-response fixed point. The figure follows its VI representation through a sign-equivalent transformation to two methods for studying uniqueness.
\end{frame}


\begin{frame}{Example 1: a two-player game}
\compactmath
Two players choose $x_i\geq0$, so $K=\mathbb R_+^2$.
\begin{block}{Payoffs}
\[
\begin{aligned}
 u_1(x)&=x_1\Bigl(1+\tfrac12x_2\Bigr)-\tfrac12x_1^2,\\
 u_2(x)&=4\Bigl[x_2\Bigl(1+\tfrac12x_1\Bigr)-\tfrac12x_2^2\Bigr].
\end{aligned}
\]
\end{block}
Each payoff is strictly concave in the player's own action.
\begin{block}{Best responses}
\[
 B_1(x_2)=1+\tfrac12x_2,\qquad
 B_2(x_1)=1+\tfrac12x_1.
\]
\end{block}
The factor $4$ changes player 2's payoff units, but leaves the best response unchanged.
\end{frame}


\begin{frame}{Example 1: the VI and its transformation}
\compactmath
For a candidate $x^*\in K$, use the negative marginal payoffs:
\[
\begin{gathered}
 F_i=-\partial_i u_i,\qquad
 F(x)=\begin{pmatrix}x_1-1-\tfrac12x_2\\4(x_2-1-\tfrac12x_1)\end{pmatrix}.\\[2pt]
 x^*\in\mathrm{NE}
 \quad\Longleftrightarrow\quad
 \langle y-x^*,F(x^*)\rangle\geq0\quad\text{for every }y\in K.
\end{gathered}
\]
\begin{block}{SET: divide the second component by $4$}
\[
 \widetilde F(x)=\operatorname{diag}(1,\tfrac14)F(x)
 =\begin{pmatrix}x_1-1-\tfrac12x_2\\x_2-1-\tfrac12x_1\end{pmatrix}.
\]
\end{block}
Positive scaling preserves every component's sign on $K$. Since $K$ is rectangular,
\[
 \operatorname{Sol}(K,F)=\operatorname{Sol}(K,\widetilde F).
\]
\end{frame}


\begin{frame}{Example 1: the two uniqueness routes}
\compactmath
\setlength{\belowdisplayskip}{3pt}
The transformed system has a symmetric derivative:
\[
 H=D\widetilde F=\begin{pmatrix}1&-\tfrac12\\-\tfrac12&1\end{pmatrix},
 \qquad\lambda(H)\in\{\tfrac12,\tfrac32\}.
\]
\begin{columns}[T,onlytextwidth]
\column{0.40\textwidth}
\textbf{$P$-property method}\par
The principal minors of $H$ are $1,1,\tfrac34>0$.
\medskip\par
The $P$-matrix condition gives at most one VI solution.
\column{0.56\textwidth}
\textbf{Potential method}
\[
\begin{aligned}
 \Phi(x)={}&\tfrac12(x_1^2+x_2^2)\\
           &-\tfrac12x_1x_2-x_1-x_2.
\end{aligned}
\]
$\nabla\Phi=\widetilde F$ and $\nabla^2\Phi=H\succ0$.
The VI minimizes the strictly convex $\Phi$ over $K$.
\end{columns}
\[
 \widetilde F(x^*)=0
 \quad\Longrightarrow\quad
 \boxed{x_1^*=x_2^*=2}.
\]
This feasible zero proves existence. The equilibrium is \alert{unique}.
\end{frame}


\begin{frame}{Example 2: different nonlinear best responses}
\compactmath
Two players choose $x_i\geq0$, with $K=\mathbb R_+^2$.
\begin{block}{Payoffs}
\[
\begin{aligned}
 u_1(x)&=x_1(1+x_2)-x_1^2-\tfrac13x_1^3,\\
 u_2(x)&=x_2\ln(2+x_1)-\tfrac12x_2^2.
\end{aligned}
\]
\end{block}
Each payoff is strictly concave in the player's own action.
\begin{block}{Best responses}
\[
 B_1(x_2)=\sqrt{2+x_2}-1,\qquad
 B_2(x_1)=\ln(2+x_1).
\]
\end{block}
Every equilibrium is interior because both best responses are positive on $K$.
\end{frame}


\begin{frame}{Example 2: a nonlinear sign-equivalent transformation}
\compactmath
The negative marginal payoffs define $F_i=-\partial_i u_i$.
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{Original VI}
\[
 F(x)=\begin{pmatrix}
 x_1^2+2x_1-1-x_2\\
 x_2-\ln(2+x_1)
 \end{pmatrix}.
\]
\[
 \partial_2F_1=-1,\quad
 \partial_1F_2=-\frac1{2+x_1}.
\]
\column{0.48\textwidth}
\textbf{After SET}
\[
 \widetilde F(x)=\begin{pmatrix}
 x_1^2+2x_1-1-x_2\\
 e^{x_2}-2-x_1
 \end{pmatrix}.
\]
\[
 \partial_2\widetilde F_1=\partial_1\widetilde F_2=-1.
\]
\end{columns}
\medskip
The exponential is strictly increasing:
\[
 x_2\gtreqless\ln(2+x_1)
 \quad\Longleftrightarrow\quad
 e^{x_2}\gtreqless2+x_1.
\]
All component signs agree on the rectangular domain $K$. Thus
\[
 \mathrm{NE}=\operatorname{Sol}(K,F)=\operatorname{Sol}(K,\widetilde F).
\]
\end{frame}


\begin{frame}{Example 2: a convex equilibrium problem}
\compactmath
The transformed operator is a gradient, $\widetilde F=\nabla\Phi$, where
\[
 \Phi(x)=\tfrac13x_1^3+x_1^2-x_1-x_1x_2+e^{x_2}-2x_2.
\]
For every $x\in K$,
\[
 \nabla^2\Phi(x)=
 \begin{pmatrix}2+2x_1&-1\\-1&e^{x_2}\end{pmatrix}
 \succeq
 \begin{pmatrix}2&-1\\-1&1\end{pmatrix}
 \succ0.
\]
\begin{block}{Unique equilibrium}
Strong convexity and coercivity give a unique minimizer on $K$.
\end{block}
SET exposes a convex optimization representation of the same equilibrium problem.
\end{frame}



\begin{frame}{Individual optimality and the VI}
\compactmath
\setcounter{theorem}{0}
\begin{definition}[Well-behaved game]
A normal-form game $\Gamma=(u_i,K_i)_{i\in N}$ is \alert{well-behaved} if, for every player $i$:
\begin{enumerate}
 \item $K_i\subseteq\R^{m_i}$ is closed and convex.
 \item $u_i$ is twice continuously differentiable ($C^2$).
 \item $u_i(\cdot,x_{-i})$ is concave on $K_i$ for every feasible $x_{-i}$.
\end{enumerate}
\end{definition}
\setcounter{theorem}{0}
\begin{lemma}[Nash equilibrium and the VI]
For a well-behaved game, set $K=\prod_i K_i$ and $F_i=-\nabla_{x_i}u_i$.
Then, for $x^*\in K$,
\[
 x^*\in\mathrm{NE}
 \quad\Longleftrightarrow\quad
 \langle y-x^*,F(x^*)\rangle\geq0\quad\text{for every }y\in K.
\]
\end{lemma}
\end{frame}


\begin{frame}{The VI as familiar boundary conditions}
\compactmath
\setcounter{theorem}{3}
\begin{lemma}[Part (i): rectangular domains]
Let $K=\prod_i[a_i,b_i]$, with $-\infty\leq a_i<b_i\leq+\infty$.
For $x^*\in K$,
\[
\begin{gathered}
 \langle y-x^*,F(x^*)\rangle\geq0\quad\text{for every }y\in K
 \\[2pt]
 \Updownarrow
 \\[2pt]
 \left.
 \begin{aligned}
  x_i^*=a_i &\quad\Longrightarrow\quad F_i(x^*)\geq0 &&\text{(lower bound)},\\[3pt]
  a_i<x_i^*<b_i &\quad\Longrightarrow\quad F_i(x^*)=0 &&\text{(interior)},\\[3pt]
  x_i^*=b_i &\quad\Longrightarrow\quad F_i(x^*)\leq0 &&\text{(upper bound)}.
 \end{aligned}
 \right\}\quad\text{for every }i.
\end{gathered}
\]
\end{lemma}
With $F_i=-\partial_i u_i$, these are the familiar constrained first-order conditions.
\medskip\par
\alert{Only the sign of each component matters, including at the boundary.}
\end{frame}


\begin{frame}{Sign equivalence on rectangular domains}
\small\compactmath
\setcounter{theorem}{1}
\begin{definition}[Sign equivalence]
$F$ and $\widetilde F$ are sign-equivalent on $K$ if
$\sgn F_i(x)=\sgn\widetilde F_i(x)$ for every $i$ and every $x\in K$.
\end{definition}
\setcounter{theorem}{0}
\begin{theorem}[Part (ii): rectangular domains]
If $K=\prod_i[a_i,b_i]$ and $F,\widetilde F$ are sign-equivalent on $K$, then
\[
 \Sol(K,F)=\Sol(K,\widetilde F).
\]
\end{theorem}
\end{frame}


\begin{frame}{Two constructions of a sign-equivalent operator}
\compactmath
\setcounter{proposition}{0}
\begin{proposition}[SET 1 and SET 2]
Finite compositions of the following componentwise operations preserve signs.
\begin{enumerate}
\item \textbf{Positive rescaling:}
\[
 \widetilde F_i(x)=\alpha_i(x)F_i(x),\qquad \alpha_i(x)>0\quad\text{on all of }K.
\]
\item \textbf{An increasing transformation:} if $F_i=A_i-B_i$ and $h_i$ is strictly increasing,
\[
 \widetilde F_i=h_i(A_i)-h_i(B_i)
 \quad\Longrightarrow\quad \sgn\widetilde F_i=\sgn F_i.
\]
\end{enumerate}
For strictly decreasing $h_i$, reverse the order: $h_i(B_i)-h_i(A_i)$.
\end{proposition}
The rescaling factor may depend on the full profile, as in oligopoly pricing.
Our Example 2 uses $h(t)=e^t$. Logs and inverses require valid domains.
\end{frame}


\begin{frame}{The role of rectangularity: a counterexample}
\compactmath
Consider a VI with a shared constraint rather than separate action intervals:
\[
 K=\{x\geq0:x_1+x_2\leq1\},\qquad
 F=(-1,-1),\qquad \widetilde F=(-2,-1).
\]
Both operators have the same component signs at every feasible point.
For a constant operator, the VI minimizes its linear objective over $K$.
\[
 \Sol(K,F)=\{x\geq0:x_1+x_2=1\},\qquad
 \Sol(K,\widetilde F)=\{(1,0)\}.
\]
\begin{block}{Discussion}
Why does changing the relative weight of the two coordinates matter when they
compete for a shared resource?
\end{block}
This is a coupled-constraint VI illustration. Its domain is not a product of intervals.
\end{frame}


\begin{frame}{The $P$-condition and constrained uniqueness}
\compactmath
A useful global $P$-condition is: for every $x\ne y$ in $K$,
\[
 \max_i (x_i-y_i)\bigl[F_i(x)-F_i(y)\bigr]>0.
\]
Suppose $x$ and $y$ both solve the VI on a rectangle.
Feasible one-coordinate deviations give, for every $i$,
\[
 (x_i-y_i)F_i(x)\leq0,\qquad
 (x_i-y_i)F_i(y)\geq0.
\]
Subtracting implies
\[
 (x_i-y_i)\bigl[F_i(x)-F_i(y)\bigr]\leq0\quad\text{for every }i.
\]
This contradicts the $P$-condition unless $x=y$.
The argument covers constrained solutions as well as interior zeros.
\end{frame}


\begin{frame}{What a better representation can reveal}
\compactmath
Let $K$ be convex and $\widetilde F$ continuously differentiable.
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\setcounter{theorem}{1}
\begin{lemma}[Part (ii): $P$-property]
On a rectangle $K$, if $D\widetilde F(x)$ is a $P$-matrix for every $x\in K$,
then the VI has \alert{at most one} solution.
\end{lemma}
A $P$-matrix has positive principal minors.
\medskip\par
\textbf{Used for:} oligopoly pricing, peer effects, CES complements.
\column{0.48\textwidth}
\setcounter{theorem}{2}
\begin{lemma}[Potential]
If $D\widetilde F$ is symmetric and positive semidefinite on $K$, the VI
minimizes a convex potential with $\nabla\Phi=\widetilde F$.
If it is positive definite, there is \alert{at most one} solution.
\end{lemma}
\textbf{Used for:} network games, public goods, multiplex interactions.
\end{columns}
\medskip
Existence needs its own argument in each application.
\end{frame}


\begin{frame}{Selected Applications}
\begin{enumerate}
\item \textbf{Oligopoly pricing:} derive the markup condition and trace how the demand assumptions deliver uniqueness.
\item \textbf{Network complements:} separate the existence threshold from uniqueness and relate equilibrium to centrality.
\item \textbf{Network public goods:} construct the potential and explain how convex costs change equilibrium behavior.
\end{enumerate}
\medskip
For each model: what does SET preserve, and which additional argument delivers the economic result?
\end{frame}


\begin{frame}{Oligopoly: what price should each firm charge?}
There are $n\geq2$ firms, each selling one differentiated product.
\begin{itemize}
 \item Firm $i$ has constant unit cost $c_i$ and chooses $p_i\in[a_i,b_i]$, with $c_i\leq a_i<b_i$.
 \item Demand $Q_i(p)>0$ depends on its own price and its rivals' prices.
 \item Consumers can also choose an outside option, whose price $p_0$ is fixed.
\end{itemize}
\begin{block}{The firm's problem}
\[
 \max_{p_i\in[a_i,b_i]}\;\pi_i(p)=(p_i-c_i)Q_i(p).
\]
\end{block}
Can firms' pricing decisions be mutually consistent in more than one way,
including when a price floor or cap binds?
\end{frame}


\begin{frame}{Where demand comes from}
\compactmath
Each consumer chooses the option with the highest utility:
\[
 U_i=v_i-p_i+\varepsilon_i,\qquad i=0,1,\ldots,n.
\]
Here $v_i$ measures product appeal; $\varepsilon_i$ captures individual tastes.
Option $0$ means not buying from any firm. Normalize the mass of consumers to one.
\begin{block}{A familiar example: multinomial logit}
With independent standard extreme-value taste shocks,
\[
 Q_i(p)=\frac{e^{v_i-p_i}}
 {e^{v_0-p_0}+\sum_{j=1}^n e^{v_j-p_j}}.
\]
\end{block}
A higher own price loses customers to rivals or the outside option.
The paper's argument covers more than this logit example.
\end{frame}


\begin{frame}{An optimal price balances two effects}
\compactmath
Differentiate profit, holding rivals' prices fixed:
\[
 \frac{\partial\pi_i}{\partial p_i}
 =\underbrace{Q_i}_{\text{more revenue per customer}}
 +\underbrace{(p_i-c_i)\frac{\partial Q_i}{\partial p_i}}_{\text{lost customers}},
 \qquad \frac{\partial Q_i}{\partial p_i}<0.
\]
\begin{block}{At an interior optimum}
\[
 \underbrace{p_i-c_i}_{\text{actual markup}}
 =\underbrace{\frac{Q_i}{-\partial Q_i/\partial p_i}}_{\text{demand-implied markup }\mu_i(p)}.
\]
\end{block}
$\mu_i$ is the inverse own-price \alert{semi-elasticity}, $1/[-\partial_{p_i}\log Q_i]$.
Less price-sensitive demand supports a larger markup; rivals' prices affect this sensitivity.
\end{frame}


\begin{frame}{SET: rewrite pricing as a markup gap}
\compactmath
The original VI uses negative marginal profit, $F_i=-\partial\pi_i/\partial p_i$.
Divide by the positive number $-\partial Q_i/\partial p_i$:
\begin{block}{A sign-equivalent pricing condition}
\[
 \boxed{\widetilde F_i(p)=\frac{F_i(p)}{-\partial Q_i/\partial p_i}
       =\underbrace{p_i-c_i}_{\text{actual markup}}-\underbrace{\mu_i(p)}_{\text{desired markup}}.}
\]
\end{block}
\begin{itemize}
 \item $\widetilde F_i<0$: the firm wants to \alert{raise} its price.
 \item $\widetilde F_i>0$: the firm wants to \alert{cut} its price.
 \item $\widetilde F_i=0$: the two pricing effects balance.
\end{itemize}
This is a new representation of the marginal condition, not a new profit function.
\end{frame}


\begin{frame}{Price floors and caps: signs, not just zeros}
\compactmath
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{lll}
\toprule
Chosen price & Marginal profit & Markup gap\\
\midrule
Floor: $p_i=a_i$ & $\partial\pi_i/\partial p_i\leq0$ & $\widetilde F_i\geq0$\\
Interior & $\partial\pi_i/\partial p_i=0$ & $\widetilde F_i=0$\\
Cap: $p_i=b_i$ & $\partial\pi_i/\partial p_i\geq0$ & $\widetilde F_i\leq0$\\
\bottomrule
\end{tabular}
\end{center}
Under E1 (own-price log-concavity), these conditions characterize best responses.
Positive rescaling preserves all three cases. On $K=\prod_i[a_i,b_i]$,
\[
 \mathrm{NE}=\Sol(K,F)=\Sol(K,\widetilde F).
\]
The \alert{entire equilibrium set} is preserved.
Looking only for zeros would miss equilibria with binding bounds.
\end{frame}


\begin{frame}{The demand restrictions behind uniqueness}
\small\compactmath
Write $\ell_i=\log Q_i$. Demand is positive, $C^2$, and decreasing in own price.
\begingroup
\renewcommand{\theassumption}{E\arabic{assumption}}
\setcounter{assumption}{0}
\begin{assumption}[Own-price log-concavity]
\compactmath
 \[
 \frac{\partial^2\ell_i}{\partial p_i^2}\leq0.
 \]
\end{assumption}
\begin{assumption}[Cross-price log-supermodularity]
\compactmath
 \[
 \frac{\partial^2\ell_i}{\partial p_i\partial p_j}\geq0
 \quad(j\ne i,\ \text{including the outside option }j=0).
 \]
\end{assumption}
\begin{assumption}[Common-price-shift invariance]
\compactmath
 \[
 Q_i(p_1+t,\ldots,p_n+t,p_0+t)=Q_i(p_1,\ldots,p_n,p_0),\quad i=0,\ldots,n.
 \]
\end{assumption}
\endgroup
\end{frame}


\begin{frame}{Why the first-order pricing conditions are sufficient}
\compactmath
For $p_i>c_i$, profit is positive. Under E1,
\[
 \log\pi_i=\log(p_i-c_i)+\log Q_i.
\]
\[
 \frac{\partial^2\log\pi_i}{\partial p_i^2}
 =-\frac1{(p_i-c_i)^2}
   +\frac{\partial^2\log Q_i}{\partial p_i^2}<0.
\]
Strict log-concavity makes the pricing incentive cross zero at most once.
Thus the interior and endpoint signs characterize a best response.
\begin{block}{The cost-price endpoint}
At $p_i=c_i$, marginal profit equals $Q_i>0$.
A feasible higher price gives positive profit, so cost pricing cannot be optimal.
\end{block}
The argument uses log-profit concavity. Profit itself may fail to be concave.
\end{frame}


\begin{frame}{E1 and E2: derivatives of the desired markup}
\compactmath
Use $\ell_i=\log Q_i$ and $\mu_i=-1/\partial_{p_i}\ell_i$. Then
\[
 \frac{\partial\mu_i}{\partial p_j}
 =\frac{\partial^2\ell_i/\partial p_i\partial p_j}
        {(\partial\ell_i/\partial p_i)^2}.
\]
\begin{itemize}
 \item E1: $\partial_{p_i}\mu_i\leq0$. A higher own price weakly lowers the desired markup.
 \item E2: $\partial_{p_j}\mu_i\geq0$ for $j\ne i$, including $j=0$.
 A higher rival price weakly raises the desired markup.
\end{itemize}
Since $\widetilde F_i=p_i-c_i-\mu_i$, its Jacobian satisfies
\[
 J_{ii}=1-\partial_{p_i}\mu_i\geq1,\qquad
 J_{ij}=-\partial_{p_j}\mu_i\leq0\quad(j\ne i).
\]
E1 controls the own effect. E2 controls the cross effects.
\end{frame}


\begin{frame}{E3: relative prices and the outside option}
\compactmath
Common-price-shift invariance also holds for the desired markup.
Differentiating with respect to the common shift gives
\[
 \sum_{j=0}^n\frac{\partial\mu_i}{\partial p_j}=0.
\]
Using the nonpositive cross effects, the firms' price Jacobian satisfies
\[
\begin{aligned}
 \sum_{j=1}^n J_{ij}
 =1-\sum_{j=1}^n\frac{\partial\mu_i}{\partial p_j}
 &=1+\frac{\partial\mu_i}{\partial p_0}\geq1,\\[4pt]
 J_{ii}-\sum_{j\ne i}|J_{ij}|&=\sum_{j=1}^nJ_{ij}\geq1.
\end{aligned}
\]
The outside-option derivative bounds the combined strategic cross effects.
\end{frame}


\begin{frame}{A direct uniqueness proof for the pricing VI}
\compactmath
Take distinct $p,q\in K$, write $d=p-q$, and choose $i$ with
$|d_i|=\|d\|_\infty>0$.
The segment $q+td$ stays in the price rectangle.
\[
\begin{aligned}
 d_i\bigl[\widetilde F_i(p)-\widetilde F_i(q)\bigr]
 &=\int_0^1 d_i\sum_j J_{ij}(q+td)d_j\,dt\\[3pt]
 &\geq d_i^2\int_0^1\left[J_{ii}(q+td)-\sum_{j\ne i}|J_{ij}(q+td)|\right]dt\\[3pt]
 &\geq d_i^2>0.
\end{aligned}
\]
This proves the global $P$-condition. The earlier coordinate argument therefore
rules out two VI solutions, including solutions with binding price bounds.
\medskip
\textbf{Discussion.} Where did E1, E2, E3, and rectangularity enter this proof?
\end{frame}


\begin{frame}{Existence and uniqueness in the pricing game}
\compactmath
\setcounter{proposition}{3}
\begin{proposition}[Oligopoly pricing]
Under E1--E3, the pricing game on $\prod_i[a_i,b_i]$ with $a_i\geq c_i$
has \alert{at most one} pure-strategy Nash equilibrium.
\end{proposition}
The proof uses strict row diagonal dominance of the transformed markup gap.
\begin{block}{Existence on finite price intervals}
Each firm has a unique best response on its compact interval.
Continuous profits make this best response continuous.
The joint best-response map is a self-map of the compact price box, so Brouwer gives a fixed point.
\end{block}
Together, these arguments give \alert{exactly one equilibrium} on the finite box.
An unbounded price domain requires a separate existence argument.
\end{frame}


\begin{frame}{Verification of the logit demand assumptions}
\compactmath
Let $Q_0$ be the outside-option share and $\ell_i=\log Q_i$.
For unit-slope logit,
\[
 \partial_{p_i}\ell_i=-(1-Q_i),\qquad
 \partial^2_{p_ip_i}\ell_i=-Q_i(1-Q_i)\leq0.
\]
For every other option, including the outside option,
\[
 \partial^2_{p_ip_j}\ell_i=Q_iQ_j\geq0\qquad(j\ne i,\ j=0\text{ allowed}).
\]
Adding $t$ to all prices multiplies every logit weight by $e^{-t}$.
The common factor cancels from each share, verifying E3.
\begin{block}{Discussion}
Which parts of the uniqueness proof depend on logit's formula, and which depend
only on E1--E3?
\end{block}
\end{frame}


\begin{frame}{A two-firm logit example: solve for prices}

 Q_i(p)=\frac{e^{-p_i}}{1+e^{-p_1}+e^{-p_2}},\qquad

\begin{frame}{What SET changes in the uniqueness test}
\small\compactmath
Keep the two-firm logit example, with zero costs and deterministic appeal.

\begin{frame}{CES peer effects on an unnormalized network}
\small\compactmath
\[
 u_i(x)=a_ix_i-\frac12x_i^2+
 \delta x_i\Bigl(\sum\nolimits_jg_{ij}x_j^\beta\Bigr)^{1/\beta}.
\]
\begin{block}{Best response}
\[
 T_i(x)=a_i+\delta S_i(x),\qquad
 S_i(x)=\Bigl(\sum\nolimits_jg_{ij}x_j^\beta\Bigr)^{1/\beta}.
\]
\end{block}
\begin{itemize}
 \item $a_i>0$, $\delta>0$, $\beta>0$, and $x_i\geq0$.
 \item $G$ is an undirected adjacency matrix, without self-loops.
 \item We use a \alert{connected, nonempty} network throughout this CES discussion.
\end{itemize}
\end{frame}

\begin{frame}{What the CES parameter changes}
\small\compactmath
For positive neighbor actions,
\[
 \frac{\partial S_i}{\partial x_j}
 =g_{ij}x_j^{\beta-1}S_i(x)^{1-\beta}.
\]
For two linked neighbors $j$ and $k$,
\[
 \frac{\partial S_i/\partial x_j}{\partial S_i/\partial x_k}
 =\left(\frac{x_j}{x_k}\right)^{\beta-1}.
\]
\begin{block}{Relative peer influence}
$\beta>1$ favors high-action peers at the margin.
$0<\beta<1$ places relatively more marginal weight on low-action peers.
\end{block}
Changing $\beta$ also changes the scale of this unnormalized aggregate.
\end{frame}


\begin{frame}{CES complements: the logarithmic system}
\small\compactmath
Every equilibrium satisfies $x_i=T_i(x)\geq a_i>0$.
The change of variables $z_i=\log x_i$ therefore covers the entire equilibrium set.
\[
 R_i(x)=x_i-T_i(x),\qquad
 H_i(z)=z_i-\log T_i(e^z).
\]
\begin{block}{SET on positive quantities}
Because the logarithm is strictly increasing,
\[
 \operatorname{sign}R_i(e^z)
 =\operatorname{sign}\!\left[\log e^{z_i}-\log T_i(e^z)\right]
 =\operatorname{sign}H_i(z).
\]
Thus $R(x)=0$ and $H(\log x)=0$ describe the same equilibria.
\end{block}
The next step uses the homogeneity of $S_i$ to study the derivative of $H$.
\end{frame}

\begin{frame}{CES uniqueness: the derivative and the proof}
\small\compactmath
Write $L_i(z)=\log T_i(e^z)$, $W=DL$, and $DH=I-W$. Direct differentiation gives
\[
 W_{ij}(z)=
 \frac{\delta S_i(e^z)}{a_i+\delta S_i(e^z)}
 \frac{g_{ij}e^{\beta z_j}}{\sum_k g_{ik}e^{\beta z_k}},
 \qquad \sum_jW_{ij}(z)<1.
\]
\begin{block}{Two distinct finite roots are impossible}
If $H(z)=H(w)=0$, then
\[
 z-w=\overline W(z-w),\qquad
 \overline W=\int_0^1W\!\left(w+t(z-w)\right)\,dt.
\]
On this compact segment, the row sums have a common bound $\alpha<1$.
Hence $\lVert z-w\rVert_\infty\leq\alpha\lVert z-w\rVert_\infty$, so $z=w$.
\end{block}
\textbf{Discussion:} where does $a_i>0$ enter? The argument gives at most one equilibrium, without a uniform contraction bound on all of $\R^n$.
\end{frame}


\begin{frame}{The spectral existence threshold}
\small\compactmath
\setcounter{theorem}{3}
\begin{theorem}[CES equilibrium]
Let $Ge=\lambda_{\max}(G)e$, $e\gg0$, and $\bar\delta=\lambda_{\max}(G)^{-1/\beta}$.
\begin{itemize}
 \item[(i)] $0<\delta<\bar\delta$: a unique positive equilibrium exists, monotone in $a$ and $\delta$.
 \item[(ii)] As $\delta\uparrow\bar\delta$, all actions diverge and
 $x_i^*/x_j^*\to(e_i/e_j)^{1/\beta}$.
 \item[(iii)] $\delta>\bar\delta$: no finite positive equilibrium exists.
\end{itemize}
\end{theorem}
The appendix proof also excludes $\delta=\bar\delta$.
\setcounter{theorem}{1}
\begin{example}[Paper example: regular network]
For common degree $d\geq1$, $a_i=a>0$, and $\delta<\bar\delta$,
\[
 x_i^*=\frac{a}{1-\delta d^{1/\beta}},\qquad
 \bar\delta=d^{-1/\beta}.
\]
\end{example}
\end{frame}

\begin{frame}{CES existence: a Perron upper bound}
\small\compactmath
Let $Ge=\lambda e$, with $e\gg0$ and $\lambda=\lambda_{\max}(G)$.
For $q_i=e_i^{1/\beta}$, homogeneity gives
\[
 S_i(q)=\lambda^{1/\beta}q_i,\qquad
 T_i(Mq)=a_i+\delta M\lambda^{1/\beta}q_i.
\]
\begin{block}{An invariant rectangle when $\delta<\bar\delta$}
Choose
\[
 M\geq\max_i\frac{a_i}{q_i(1-\delta\lambda^{1/\beta})},
 \qquad \overline x=Mq.
\]
Then $a\leq T(a)$, $T(\overline x)\leq\overline x$, and $T$ is increasing.
Starting from $x^{(0)}=a$, iteration increases and remains below $\overline x$.
\end{block}
The bounded sequence converges. Continuity gives $x^*=T(x^*)$.
The preceding uniqueness argument makes this the only equilibrium.
\end{frame}

\begin{frame}{CES non-existence: why the threshold is necessary}
\small\compactmath
Suppose a finite equilibrium exists. Since $a\gg0$ and $\beta>0$,
\[
 (x-a)^\beta=\delta^\beta Gx^\beta
 \quad\text{and}\quad
 (x-a)^\beta<x^\beta.
\]
Powers and inequalities here apply coordinate by coordinate.
\begin{block}{A Perron-weighted contradiction}
Because $G$ is symmetric, $e^\top G=\lambda_{\max}e^\top$. Thus
\[
 \delta^\beta\lambda_{\max}e^\top x^\beta
 =e^\top(x-a)^\beta<e^\top x^\beta,
 \qquad
 \delta<\lambda_{\max}^{-1/\beta}=\bar\delta.
\]
No finite positive equilibrium can exist at or above $\bar\delta$.
\end{block}
\textbf{Discussion:} uniqueness of a system's roots does not guarantee that a root exists.
\end{frame}

\begin{frame}{Comparative statics below the threshold}
\setcounter{theorem}{3}
\begin{theorem}[Part (i): comparative statics]
Within the region where a finite equilibrium exists,
\[
 a'\geq a\quad\Longrightarrow\quad x^*(a',\delta,\beta)\geq x^*(a,\delta,\beta),
\]
\[
 \delta'\geq\delta\quad\Longrightarrow\quad x^*(a,\delta',\beta)\geq x^*(a,\delta,\beta).
\]
\end{theorem}
\begin{itemize}
 \item A higher individual baseline or stronger peer interaction raises equilibrium effort across the network.
 \item Equilibrium payoff is $u_i(x^*)=\tfrac12(x_i^*)^2$: changes in effort also change welfare.
\end{itemize}
\end{frame}

\begin{frame}{Bonacich centrality as the linear special case}
When $\beta=1$ and $0<\delta<1/\lambda_{\max}(G)$,
\[
 x=a+\delta Gx.
\]
\begin{block}{The network multiplier}
\[
 (I-\delta G)x=a,\qquad
 x^*=(I-\delta G)^{-1}a.
\]
\end{block}
\begin{itemize}
 \item $a=\mathbf1$: a standard Katz--Bonacich centrality vector.
 \item Heterogeneous $a$: baseline-weighted centrality.
 \item In the notation $(I-\delta G)x=b$, the vector $b$ is this same baseline.
\end{itemize}
\end{frame}

\begin{frame}{The network multiplier in eigenvector coordinates}
\small\compactmath
Since $G$ is symmetric, choose orthonormal eigenvectors $v_1,\ldots,v_n$,
with $Gv_k=\lambda_kv_k$, $v_1=e$, and $\lambda_1=\lambda_{\max}$.
\[
 a=\sum_k(v_k^\top a)v_k,
 \qquad
 x^*=(I-\delta G)^{-1}a
 =\sum_k\frac{v_k^\top a}{1-\delta\lambda_k}v_k.
\]
Each eigenvalue determines the amplification of its corresponding direction.
\begin{block}{The leading direction near the threshold}
Connectedness makes $\lambda_1$ simple, and $e^\top a>0$. Therefore
\[
 (1-\delta\lambda_1)x^*\longrightarrow(e^\top a)e,
 \qquad \frac{x^*}{\lVert x^*\rVert_2}\longrightarrow e.
\]
\end{block}
\textbf{Discussion:} baseline heterogeneity remains in $e^\top a$. Why does it disappear after normalization?
\end{frame}

\begin{frame}{The nonlinear Perron limit: proof}
\small\compactmath
As $\delta\uparrow\bar\delta$, bounded actions would yield an equilibrium at the threshold by continuity. Connectedness propagates divergence through
$x_i\geq\delta x_j$ whenever $g_{ij}=1$. Thus every $a_i/x_i\to0$.
\[
 y=x^\beta,\qquad q=\frac{y}{\lVert y\rVert_1},\qquad
 \frac{(x-a)^\beta}{\lVert x^\beta\rVert_1}=\delta^\beta Gq.
\]
\begin{block}{Every normalized subsequence has the same limit}
For any convergent subsequence $q\to\widehat q$,
\[
 \widehat q=\bar\delta^\beta G\widehat q,\qquad
 \widehat q\geq0,\qquad \lVert\widehat q\rVert_1=1.
\]
Perron uniqueness gives $\widehat q=e/\lVert e\rVert_1$.
Hence the full normalized sequence converges to that vector.
\end{block}
Taking $\beta$th roots yields the limiting ratios on the next slide.
\end{frame}

\begin{frame}{Eigenvector centrality at the critical limit}
\small\compactmath
For the positive unit Perron vector $e$,
\[
 Ge=\lambda_{\max}(G)e,\qquad e\gg0.
\]
\setcounter{theorem}{3}
\begin{theorem}[Part (ii): limiting ratios]
\[
 \lim_{\delta\uparrow\bar\delta}\frac{x_i^*}{x_j^*}
 =\left(\frac{e_i}{e_j}\right)^{1/\beta}.
\]
\end{theorem}
\begin{itemize}
 \item $\beta=1$: $x^*/\lVert x^*\rVert_2\to e$.
 \item General $\beta>0$: the vector $(x^*)^\beta$ approaches the Perron direction.
 \item Baseline heterogeneity affects levels, but disappears from these limiting ratios.
\end{itemize}
\end{frame}

\begin{frame}{Network public goods with nonlinear costs}
\small\compactmath
On an undirected network, player $i$ chooses $x_i\geq0$ and receives
\[
 u_i(x)=b\!\left(x_i+\delta\sum_jg_{ij}x_j\right)-c(x_i),
 \qquad 0<\delta\leq1.
\]
\setcounter{assumption}{1}
\begin{assumption}[Benefits, costs, and a marginal crossing]
\[
 b'>0,\qquad b''<0,\qquad c'(x)>0\ (x>0),\qquad c''\geq0,
\]
\[
 b'(0)>c'(0),\qquad
 \lim_{x\to\infty}b'(x)<\lim_{x\to\infty}c'(x).
\]
\end{assumption}
The positive autarky effort $k^*$ uniquely solves $b'(k^*)=c'(k^*)$.

The benchmark of Bramoull\'e and Kranton (2007) has $c(x)=x$ and $\delta=1$.
\end{frame}

\begin{frame}{Best responses, corners, and existence}
\small
Let $t_i=\delta\sum_jg_{ij}x_j$ be the public good supplied by neighbors.
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{If $x_i>0$}
The agent balances marginal benefit and cost:
\[
 b'(x_i+t_i)=c'(x_i).
\]
\end{block}
\column{0.48\textwidth}
\begin{block}{If $x_i=0$}
The first unit is not worthwhile when
\[
 b'(t_i)\leq c'(0).
\]
\end{block}
\end{columns}
\medskip
Strict concavity gives a unique continuous best response, at most $k^*$.
The best-response mapping sends $[0,k^*]^n$ into itself.
Brouwer's theorem therefore gives existence, before any spectral restriction.
\end{frame}


\begin{frame}{The slope of the best response}
\small\compactmath
Write $\theta(t)=\argmax_{x\geq0}\{b(x+t)-c(x)\}$.
At an interior choice, differentiate $b'(\theta(t)+t)=c'(\theta(t))$:
\[
 b''(\theta+t)\,[\theta'(t)+1]=c''(\theta)\theta'(t),
 \qquad
 \theta'(t)=\frac{b''(\theta+t)}{c''(\theta)-b''(\theta+t)}.
\]
\textbf{Cost curvature.}
Because $b''<0$ and $c''\geq0$, $-1\leq\theta'(t)<0$ at an interior choice.
\begin{itemize}
 \item Linear costs give one-for-one crowding out at interior choices.
 \item Positive $c''$ reduces the magnitude of this local response.
 \item At zero, further neighbor provision keeps effort at zero.
\end{itemize}
\textbf{Discussion:} does a slope bound for one best response prove uniqueness of the network fixed point?
\end{frame}

\begin{frame}{The inverse marginal benefit: a valid domain}
\small\compactmath
Before using $(b')^{-1}$, its argument must lie in the range of $b'$.
Let $d_{\max}$ be the largest degree and define
\[
 B=\delta d_{\max}k^*,\qquad \ell=\theta(B),\qquad K=[\ell,k^*]^n.
\]
For $x\in[0,k^*]^n$, $0\leq t_i\leq B$, so every best response lies in $K$.
Restricting the game to $K$ retains its entire equilibrium set.
\begin{block}{A finite inverse on the equilibrium-containing box}
The first-order or corner condition at $\theta(B)$ gives, for $u\in[\ell,k^*]$,
\[
 \lim_{s\to\infty}b'(s)<c'(u)
 \leq c'(k^*)=b'(k^*)<b'(0).
\]
Thus $h(u)=(b')^{-1}(c'(u))$ is well-defined on this interval.
\end{block}
When $c'(0)=0$, $\ell>0$. No finite inverse value at zero is needed.
\end{frame}

\begin{frame}{Substitutes: SET and a symmetric derivative}
\small\compactmath
For $s_i=x_i+\delta\sum_jg_{ij}x_j$, the decreasing function $b'$ gives
\[
 F_i=c'(x_i)-b'(s_i)
 \quad\sim_{\rm sign}\quad
 \widetilde F_i=s_i-h(x_i),\qquad x\in K.
\]
For $j\ne i$, the derivatives are
\[
 \frac{\partial F_i}{\partial x_j}=-\delta g_{ij}b''(s_i),
 \qquad
 \frac{\partial\widetilde F_i}{\partial x_j}=\delta g_{ij}
 =\frac{\partial\widetilde F_j}{\partial x_i}.
\]
The original derivative is generally asymmetric. The transformed system has the minimization potential
\[
 \Psi(x)=\sum_i\left[\frac{x_i^2}{2}-\int_{\ell}^{x_i}h(u)\,du\right]
 +\frac{\delta}{2}x^\top Gx,
 \qquad \nabla\Psi=\widetilde F.
\]
SET preserves the constrained equilibria on $K$, including any zero-effort corners when $\ell=0$.
\end{frame}

\begin{frame}{Substitutes: the spectral uniqueness proof}
\small\compactmath
Differentiate the inverse relation $b'(h(u))=c'(u)$:
\[
 h'(u)=\frac{c''(u)}{b''(h(u))}\leq0,
 \qquad D^2\Psi=I+\delta G-\operatorname{diag}(h'(x_i)).
\]
For any vector $v$, the Rayleigh bound gives
\[
 v^\top D^2\Psi\,v
 \geq[1+\delta\lambda_{\min}(G)]\lVert v\rVert_2^2.
\]
\setcounter{theorem}{4}
\begin{theorem}[Public-good equilibrium]
Under Assumption 2, if $1+\delta\lambda_{\min}(G)>0$, the paper's maximization
potential $\phi^{\mathrm{BKC}}$ is strictly concave and the game has a unique Nash equilibrium.
\end{theorem}
On $K$, $\Psi=-\phi^{\mathrm{BKC}}+\text{constant}$, using a finite integration reference.
Strict convexity gives at most one solution; earlier compactness gives existence.
The extra curvature $-h'$ can give uniqueness beyond the bound, as the complete-network example shows.
\end{frame}


\begin{frame}{Zero marginal cost at zero implies full participation}
\small\compactmath
Maintain Assumption 2, add strict convexity of $c$, and set $\delta=1$.
\setcounter{assumption}{2}
\begin{assumption}[Zero marginal cost at zero]
$b'(0)>c'(0)=0$.
\end{assumption}
\setcounter{proposition}{6}
\begin{proposition}[Part (i): full participation]
Every Nash equilibrium is interior: $x_i^*>0$ for every player.
\end{proposition}
\textbf{Proof (independent of uniqueness).} At any finite profile with $x_i=0$,
\[
 \left.\frac{\partial u_i}{\partial x_i}\right|_{x_i=0}
   =b'\!\Bigl(\sum\nolimits_jg_{ij}x_j\Bigr)-c'(0)>0.
\]
All equilibrium conditions therefore hold with equality:
\[
 b'\!\Bigl(x_i^*+\sum\nolimits_jg_{ij}x_j^*\Bigr)=c'(x_i^*).
\]
\end{frame}

\begin{frame}{Complete networks: convex costs remove the continuum}
\small\compactmath
On a complete network with $n\geq2$ and $\delta=1$, write $S=\sum_jx_j$.
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\setcounter{proposition}{6}
\begin{proposition}[Part (iii)]
Under Assumptions 2--3 and strictly convex costs, the equilibrium is unique:
\[
 x_i^*=\bar x,\qquad b'(n\bar x)=c'(\bar x).
\]
\end{proposition}
At any equilibrium,
\[
 c'(x_i)=b'(S)\quad\text{for all }i.
\]
Increasing $c'$ forces equal effort; the decreasing function $b'(nz)-c'(z)$ crosses zero once.
\column{0.48\textwidth}
\begin{block}{Linear costs, $c(x)=x$}
With $b'(k^*)=1$, every profile satisfying
\[
 x_i\geq0,\qquad \sum_ix_i=k^*
\]
is an equilibrium.

The continuum includes both equal sharing and a single specialist.
\end{block}
\end{columns}
Here $\lambda_{\min}(G)=-1$. Uniqueness with strictly convex costs holds even
though $1+\delta\lambda_{\min}(G)=0$.
\end{frame}

\begin{frame}{Network structure determines effort rankings}
\small\compactmath
Maintain Assumptions 2--3, homogeneous $b,c$, strictly convex $c$, and $\delta=1$.
Let $N_i$ and $\overline N_i=N_i\cup\{i\}$ denote open and closed neighborhoods.
\setcounter{proposition}{6}
\begin{proposition}[Part (ii): neighborhood inclusion]
At every equilibrium,
\[
 N_i\subseteq N_j\ \text{ or }\quad
 \overline N_i\subseteq\overline N_j
 \quad\Longrightarrow\quad x_i^*\geq x_j^*.
\]
Proper inclusion makes the effort inequality strict.
\end{proposition}
\setcounter{theorem}{1}
\begin{corollary}[Nested split graphs]
Under the same assumptions, if $G$ is an NSG, then
$d_i<d_j$ implies $x_i^*>x_j^*$ at every Nash equilibrium.
\end{corollary}
\textbf{Star, $n\geq3$.} Proposition 7(iv) gives equal leaf effort, strictly above the hub's.
Degree alone does not imply this ranking on arbitrary networks.
\end{frame}

\begin{frame}{Why the hub contributes less in a star}
\small\compactmath
Maintain Assumptions 2--3, homogeneous $b,c$, strictly convex $c$, and $\delta=1$.
\setcounter{proposition}{6}
\begin{proposition}[Part (iv): star network]
In a star with $n\geq3$, every equilibrium has common leaf effort $x_L>x_H$, where $x_H$ is the hub's effort.
\end{proposition}
\begin{enumerate}
 \item All leaves face $x_H$. Their unique best response gives one common $x_L>0$.
 \item The hub and a leaf satisfy
 \[
 c'(x_H)=b'\!\left(x_H+(n-1)x_L\right),\qquad
 c'(x_L)=b'(x_H+x_L).
 \]
 \item Since $(n-2)x_L>0$, decreasing $b'$ and increasing $c'$ imply
 \[
 c'(x_H)<c'(x_L)\quad\Longrightarrow\quad\boxed{x_H<x_L}.
 \]
\end{enumerate}
The proof applies to every equilibrium, without a uniqueness assumption.
\end{frame}

\begin{frame}{Backup: a coordinate proof of Theorem 1(ii)}
\small\compactmath
Let $K=\prod_i K_i$ and suppose $x^*\in\operatorname{Sol}(K,F)$.
For every $i$ and $t\in K_i$,
\[
 (t-x_i^*)F_i(x^*)\geq0.
\]
Sign equivalence implies the same inequality for $\widetilde F_i(x^*)$:
\[
 \begin{cases}
 F_i(x^*)=0 &\Longrightarrow\ \widetilde F_i(x^*)=0,\\
 F_i(x^*)>0 &\Longrightarrow\ t-x_i^*\geq0\text{ and }\widetilde F_i(x^*)>0,\\
 F_i(x^*)<0 &\Longrightarrow\ t-x_i^*\leq0\text{ and }\widetilde F_i(x^*)<0.
 \end{cases}
\]
Consequently, for every $x\in K$,
\[
 \langle x-x^*,\widetilde F(x^*)\rangle
 =\sum_i(x_i-x_i^*)\widetilde F_i(x^*)\geq0.
\]
Thus $\operatorname{Sol}(K,F)\subseteq\operatorname{Sol}(K,\widetilde F)$.
Interchanging the two operators gives the reverse inclusion.
\end{frame}

\begin{frame}{Backup: matrix tests in the payoff-scaling example}
\small\compactmath
For $J=\begin{pmatrix}1&-\delta\\-\kappa\delta&\kappa\end{pmatrix}$ with $\kappa>0$:
\begin{columns}[T,onlytextwidth]
\column{0.47\textwidth}
\begin{block}{P-matrix test}
The principal minors are
\[
 1,\quad\kappa,\quad\kappa(1-\delta^2).
\]
All are positive if $0<\delta<1$.
\end{block}
\column{0.47\textwidth}
\begin{block}{Strict monotonicity test}
For $S=(J+J^\top)/2$,
\[
 \det S=\kappa-\frac{(1+\kappa)^2\delta^2}{4}.
\]
$S\succ0$ iff $\delta<2\sqrt\kappa/(1+\kappa)$.
\end{block}
\end{columns}
Let $\Delta=(1-\kappa)^2+(1+\kappa)^2\delta^2$. The smallest eigenvalue is
\[
 \lambda_{\min}(S)=\frac{1+\kappa-\sqrt{\Delta}}{2}.
\]
Rescaling by $\operatorname{diag}(1,1/\kappa)$ restores the symmetric derivative
with eigenvalues $1-\delta$ and $1+\delta$.
\end{frame}

 G=\begin{pmatrix}0&1\\1&0\end{pmatrix},\qquad

Whenever $\delta\neq\pm1$, the linear system has the solution

 x=(I-\delta G)^{-1}\mathbf1=\frac{1}{1-\delta}\mathbf1.

\begin{frame}{Backup: proof of Proposition 7(iii)}
\small\compactmath
Assume homogeneous $b,c$, strictly convex $c$, $c'(0)=0$, and $\delta=1$.
On the complete network, write $S=\sum_jx_j$.
\begin{enumerate}
 \item Full participation makes every first-order condition an equality:
 \[
 c'(x_i)=b'(S)=c'(x_j)\qquad\text{for every }i,j.
 \]
 \item Strictly increasing $c'$ implies $x_i=x_j=\bar x$.
 \item The function $f(z)=b'(nz)-c'(z)$ is strictly decreasing, with
 $f(0)>0$ and an eventually negative value. Thus it has exactly one positive zero.
\end{enumerate}
\begin{block}{Why the spectral test is only sufficient}
For $n\geq2$, the complete graph has $\lambda_{\min}(G)=-1$.
At $\delta=1$ the strict spectral inequality fails, but the argument above proves uniqueness.
\end{block}
\end{frame}
