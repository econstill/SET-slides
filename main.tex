\begin{frame}{Equilibrium uniqueness and representation}
In nonlinear models, standard sufficient conditions for uniqueness
can be difficult to verify.

Some of this difficulty may come from how the equilibrium
conditions are written.

\begin{block}{The central question}
How can we transform nonlinear equilibrium to simplify analysis, without changing the equilibrium set?
\end{block}

\medskip
The paper develops \alert{sign-equivalent transformations (SET)}
to reveal useful mathematical structure in equilibrium problems.
\end{frame}


\begin{frame}{The paper's map}
\begin{center}
\includegraphics[width=0.87\textwidth]{assets/set-figure1-methodology.png}
\end{center}

\small An equilibrium is a best-response fixed point. The figure follows its VI representation through a sign-equivalent transformation to two methods for studying uniqueness.
\end{frame}


\begin{frame}{Example: different nonlinear best responses}
\compactmath
Two players choose $x_i\geq0$, with $K=\mathbb R_+^2$.
Write $a(x_1)=e^{2-x_1}>0$.
\begin{block}{Payoffs}
\[
\begin{aligned}
 u_1(x)&=x_1(1+x_2^2)-\tfrac12x_1^2,\\
 u_2(x)&=3a(x_1)^2x_2-a(x_1)x_2^2-\tfrac13x_2^3.
\end{aligned}
\]
\end{block}
Own-action curvature is $\partial_{11}u_1=-1$ and
$\partial_{22}u_2=-2(a+x_2)<0$.
\begin{block}{Best responses}
\[
 B_1(x_2)=1+x_2^2,\qquad B_2(x_1)=e^{2-x_1}.
\]
\end{block}
Player 1 responds positively; player 2 responds negatively. Both responses are positive.
\end{frame}


\begin{frame}{Example: the original VI}
\compactmath
The negative marginal payoffs define $F_i=-\partial_i u_i$.
\[
 F(x)=\begin{pmatrix}
 x_1-1-x_2^2\\
 x_2^2+2a x_2-3a^2
 \end{pmatrix}.
\]
\[
 x^*\in\mathrm{NE}
 \quad\Longleftrightarrow\quad
 \langle y-x^*,F(x^*)\rangle\geq0\quad\text{for every }y\in K.
\]
One possible uniqueness test asks whether $DF(x)$ is a $P$-matrix throughout $K$.
\begin{block}{The two-dimensional $P$-matrix test}
A matrix $J$ is a $P$-matrix if every principal minor is positive:
\[
 J_{11}>0,\qquad J_{22}>0,\qquad \det J>0.
\]
\end{block}
\end{frame}


\begin{frame}{Example: the original $P$-matrix test fails}
\compactmath
The original Jacobian is
\[
 DF(x)=\begin{pmatrix}
 1&-2x_2\\
 2a(3a-x_2)&2(a+x_2)
 \end{pmatrix}.
\]
Its diagonal entries are positive, but its determinant need not be:
\[
 \det DF(x)=2(a+x_2)+4a x_2(3a-x_2).
\]
At the interior feasible point $\widehat x=(2+\ln2,3)$, we have $a=1/2$ and
\[
 DF(\widehat x)=\begin{pmatrix}1&-6\\-3/2&7\end{pmatrix},
 \qquad \boxed{\det DF(\widehat x)=-2<0.}
\]
The original Jacobian fails the global $P$-matrix test.
This failure alone says nothing about the number of equilibria.
\end{frame}


\begin{frame}{Example: best responses are not a contraction}
\compactmath
The joint best-response map and its Jacobian are
\[
 B(x)=\begin{pmatrix}1+x_2^2\\e^{2-x_1}\end{pmatrix},\qquad
 DB(x)=\begin{pmatrix}0&2x_2\\-e^{2-x_1}&0\end{pmatrix}.
\]
At the interior equilibrium $x^*=(2,1)$,
\[
 DB(x^*)=\begin{pmatrix}0&2\\-1&0\end{pmatrix},\qquad
 \lambda=\pm i\sqrt2,\qquad \rho(DB(x^*))=\sqrt2>1.
\]
Any contraction with modulus $q<1$ would require, in its induced norm,
\[
 \rho(DB(x^*))\leq\lVert DB(x^*)\rVert\leq q<1,
\]
a contradiction. Thus $B$ is not a contraction in any fixed norm, even locally.
\end{frame}


\begin{frame}{Example: a nonlinear sign-equivalent transformation}
\compactmath
Complete the square in the second component:
\[
 F_2=x_2^2+2a x_2-3a^2=(x_2+a)^2-(2a)^2.
\]
Use the strictly increasing function $h(t)=\sqrt t$ on $(0,\infty)$:
\[
 \widetilde F_2(x)
 =\sqrt{(x_2+a)^2}-\sqrt{(2a)^2}=x_2-a.
\]
Keeping $\widetilde F_1=F_1$ gives
\[
 \widetilde F(x)=\begin{pmatrix}
 x_1-1-x_2^2\\x_2-e^{2-x_1}
 \end{pmatrix},\qquad
 \sgn\widetilde F_i(x)=\sgn F_i(x)\quad\text{on }K.
\]
Both square-root inputs are strictly positive on $K$. Rectangularity gives
\[
 \mathrm{NE}=\Sol(K,F)=\Sol(K,\widetilde F).
\]
\end{frame}


\begin{frame}{Example: the transformed $P$-matrix test}
\compactmath
The transformed Jacobian is
\[
 D\widetilde F(x)=\begin{pmatrix}
 1&-2x_2\\e^{2-x_1}&1
 \end{pmatrix}.
\]
For every $x\in K=\mathbb R_+^2$, its three principal minors satisfy
\[
 (D\widetilde F)_{11}=(D\widetilde F)_{22}=1>0,
\]
\[
 \det D\widetilde F(x)=1+2x_2e^{2-x_1}\geq1>0.
\]
Thus $D\widetilde F$ is a $P$-matrix throughout the same feasible domain.
The transformed VI has \alert{at most one} solution, so the game has at most one equilibrium.
\medskip\par
This is the $P$-matrix route in the paper's map; no potential is needed.
\end{frame}


\begin{frame}{Example: existence and the unique equilibrium}
\compactmath
Existence is explicit: at $x^*=(2,1)$,
\[
 \widetilde F(2,1)=\begin{pmatrix}2-1-1^2\\1-e^{2-2}\end{pmatrix}
 =\begin{pmatrix}0\\0\end{pmatrix}.
\]
Every best response is strictly positive, so no equilibrium lies on the boundary.
\[
 \boxed{\mathrm{NE}=\Sol(K,F)=\Sol(K,\widetilde F)=\{(2,1)\}.}
\]
\medskip
\begin{itemize}
 \item Original $DF$: the global $P$-matrix test fails.
 \item Best responses: no contraction in any fixed norm.
 \item After SET: a global $P$-matrix certificate gives uniqueness.
\end{itemize}
\end{frame}



\begin{frame}{Individual optimality and the VI}
\compactmath
\setcounter{theorem}{0}
\begin{definition}[Well-behaved game]
A normal-form game $\Gamma=(u_i,K_i)_{i\in N}$ is \alert{well-behaved} if, for every player $i$:
\begin{enumerate}
 \item $K_i\subseteq\R^{m_i}$ is closed and convex.
 \item $u_i$ is twice continuously differentiable ($C^2$).
 \item $u_i(\cdot,x_{-i})$ is concave on $K_i$ for every feasible $x_{-i}$.
\end{enumerate}
\end{definition}
\setcounter{theorem}{0}
\begin{lemma}[Nash equilibrium and the VI]
For a well-behaved game, set $K=\prod_i K_i$ and $F_i=-\nabla_{x_i}u_i$.
Then, for $x^*\in K$,
\[
 x^*\in\mathrm{NE}
 \quad\Longleftrightarrow\quad
 \langle y-x^*,F(x^*)\rangle\geq0\quad\text{for every }y\in K.
\]
\end{lemma}
\end{frame}


\begin{frame}{The VI as familiar boundary conditions}
\compactmath
\begin{lemma}
Let $K=\prod_i[a_i,b_i]$, with $-\infty\leq a_i<b_i\leq+\infty$.
For $x^*\in K$,
\[
\begin{gathered}
 \langle y-x^*,F(x^*)\rangle\geq0\quad\text{for every }y\in K
 \\[2pt]
 \Updownarrow
 \\[2pt]
 \left.
 \begin{aligned}
  x_i^*=a_i &\quad\Longrightarrow\quad F_i(x^*)\geq0 &&\text{(lower bound)},\\[3pt]
  a_i<x_i^*<b_i &\quad\Longrightarrow\quad F_i(x^*)=0 &&\text{(interior)},\\[3pt]
  x_i^*=b_i &\quad\Longrightarrow\quad F_i(x^*)\leq0 &&\text{(upper bound)}.
 \end{aligned}
 \right\}\quad\text{for every }i.
\end{gathered}
\]
\end{lemma}
With $F_i=-\partial_i u_i$, these are the familiar constrained first-order conditions.
\medskip\par
\alert{Only the sign of each component matters, including at the boundary.}
\end{frame}


\begin{frame}{Sign equivalence on rectangular domains}
\small\compactmath
\setcounter{theorem}{1}
\begin{definition}[Sign equivalence]
$F$ and $\widetilde F$ are sign-equivalent on $K$ if
$\sgn F_i(x)=\sgn\widetilde F_i(x)$ for every $i$ and every $x\in K$.
\end{definition}
\setcounter{theorem}{0}
\begin{theorem}[Part (ii): rectangular domains]
If $K=\prod_i[a_i,b_i]$ and $F,\widetilde F$ are sign-equivalent on $K$, then
\[
 \Sol(K,F)=\Sol(K,\widetilde F).
\]
\end{theorem}

\end{frame}


\begin{frame}{Two constructions of a sign-equivalent operator}
\compactmath
\setcounter{proposition}{0}
\begin{proposition}[SET 1 and SET 2]
Finite compositions of the following componentwise operations preserve signs.
\begin{enumerate}
\item \textbf{Positive rescaling:}
\[
 \widetilde F_i(x)=\alpha_i(x)F_i(x),\qquad \alpha_i(x)>0\quad\text{on all of }K.
\]
\item \textbf{An increasing transformation:} if $F_i=A_i-B_i$ and $h_i$ is strictly increasing,
\[
 \widetilde F_i=h_i(A_i)-h_i(B_i)
 \quad\Longrightarrow\quad \sgn\widetilde F_i=\sgn F_i.
\]
\end{enumerate}
For strictly decreasing $h_i$, reverse the order: $h_i(B_i)-h_i(A_i)$.
\end{proposition}
The rescaling factor may depend on the full profile, as in oligopoly pricing.
Our worked example uses $h(t)=\sqrt t$ on strictly positive arguments.
Logs and inverses require valid domains.
\end{frame}


\begin{frame}{The role of rectangularity: a counterexample}
\compactmath
Consider a VI with a shared constraint rather than separate action intervals:
\[
 K=\{x\geq0:x_1+x_2\leq1\},\qquad
 F=(-1,-1),\qquad \widetilde F=(-2,-1).
\]
Both operators have the same component signs at every feasible point.
For a constant operator, the VI minimizes its linear objective over $K$.
\[
 \Sol(K,F)=\{x\geq0:x_1+x_2=1\},\qquad
 \Sol(K,\widetilde F)=\{(1,0)\}.
\]
\begin{block}{Discussion}
Why does changing the relative weight of the two coordinates matter when they
compete for a shared resource?
\end{block}
This is a coupled-constraint VI illustration. Its domain is not a product of intervals.
\end{frame}


\begin{frame}{The $P$-condition and constrained uniqueness}
\compactmath
A useful global $P$-condition is: for every $x\ne y$ in $K$,
\[
 \max_i (x_i-y_i)\bigl[F_i(x)-F_i(y)\bigr]>0.
\]
Suppose $x$ and $y$ both solve the VI on a rectangle.
Feasible one-coordinate deviations give, for every $i$,
\[
 (x_i-y_i)F_i(x)\leq0,\qquad
 (x_i-y_i)F_i(y)\geq0.
\]
Subtracting implies
\[
 (x_i-y_i)\bigl[F_i(x)-F_i(y)\bigr]\leq0\quad\text{for every }i.
\]
This contradicts the $P$-condition unless $x=y$.
The argument covers constrained solutions as well as interior zeros.
\end{frame}


\begin{frame}{What a better representation can reveal}
\compactmath
Let $K$ be convex and $\widetilde F$ continuously differentiable.
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\setcounter{theorem}{1}
\begin{lemma}[Part (ii): $P$-property]
On a rectangle $K$, if $D\widetilde F(x)$ is a $P$-matrix for every $x\in K$,
then the VI has \alert{at most one} solution.
\end{lemma}
A $P$-matrix has positive principal minors.
\medskip\par
\textbf{Used for:} oligopoly pricing, peer effects, CES complements.
\column{0.48\textwidth}
\setcounter{theorem}{2}
\begin{lemma}[Potential]
If $D\widetilde F$ is symmetric and positive semidefinite on $K$, the VI
minimizes a convex potential with $\nabla\Phi=\widetilde F$.
If it is positive definite, there is \alert{at most one} solution.
\end{lemma}
\textbf{Used for:} network games, public goods, multiplex interactions.
\end{columns}
\medskip
Existence needs its own argument in each application.
\end{frame}


\begin{frame}{Selected Applications}
Each application follows the same comparison.
\begin{enumerate}
 \item Write the original equilibrium operator $F$ and its Jacobian.
 \item Apply a sign-equivalent transformation.
 \item Inspect the new Jacobian and state the uniqueness condition.
\end{enumerate}
\medskip
\begin{itemize}
 \item \textbf{Oligopoly pricing:} positive rescaling and a $P$-matrix.
 \item \textbf{CES network complements:} logarithms and a $P$-matrix.
 \item \textbf{Network public goods:} an inverse function and a symmetric Jacobian.
\end{itemize}
\end{frame}


\begin{frame}{Oligopoly: the original equilibrium operator}
\compactmath
Firm $i$ chooses $p_i\in[a_i,b_i]$, with $c_i\leq a_i<b_i$, and earns
\[
 \pi_i(p)=(p_i-c_i)Q_i(p),\qquad Q_i>0,\quad
 Q_{i,i}:=\frac{\partial Q_i}{\partial p_i}<0.
\]
The outside-option price $p_0$ is fixed. Let $K=\prod_i[a_i,b_i]$.
The original operator is negative marginal profit:
\[
 \boxed{F_i(p)=-Q_i(p)-(p_i-c_i)Q_{i,i}(p).}
\]
Under E1, log-profit is strictly concave for $p_i>c_i$.
At $p_i=c_i$, marginal profit is $Q_i>0$.
Hence the marginal-profit signs characterize best responses, and
\[
 \mathrm{NE}=\Sol(K,F).
\]
\end{frame}


\begin{frame}{The demand restrictions}
Write $\ell_i=\log Q_i$. Demand is positive, $C^2$, and decreasing in own price.
\begingroup
\renewcommand{\theassumption}{E\arabic{assumption}}
\setcounter{assumption}{0}
\begin{assumption}[Own-price log-concavity]
\compactmath
 \[
 \frac{\partial^2\ell_i}{\partial p_i^2}\leq0.
 \]
\end{assumption}
\begin{assumption}[Cross-price log-supermodularity]
\compactmath
 \[
 \frac{\partial^2\ell_i}{\partial p_i\partial p_j}\geq0
 \quad(j\ne i,\ \text{including the outside option }j=0).
 \]
\end{assumption}
\begin{assumption}[Common-price-shift invariance]
\compactmath
 \[
 Q_i(p_1+t,\ldots,p_n+t,p_0+t)=Q_i(p_1,\ldots,p_n,p_0),\quad i=0,\ldots,n.
 \]
\end{assumption}
\endgroup
\end{frame}


\begin{frame}{Oligopoly: the original Jacobian}
\compactmath
Write $Q_{i,j}=\partial Q_i/\partial p_j$ and
$Q_{i,ij}=\partial^2Q_i/\partial p_i\partial p_j$.
Differentiating $F_i=-Q_i-(p_i-c_i)Q_{i,i}$ gives
\[
 \boxed{
 \begin{aligned}
 (DF)_{ii}&=-2Q_{i,i}-(p_i-c_i)Q_{i,ii},\\
 (DF)_{ij}&=-Q_{i,j}-(p_i-c_i)Q_{i,ij}\qquad(j\ne i).
 \end{aligned}}
\]
The own-price demand slope contributes a positive term to the diagonal.
The second-derivative term has no such guaranteed sign.
\medskip\par
Even the diagonal entries need not be positive.
The original Jacobian therefore need not pass the global $P$-matrix test.
\end{frame}


\begin{frame}{Oligopoly: SET removes the demand-slope scale}
\compactmath
Divide each component by the positive number $-Q_{i,i}$:
\[
 \widetilde F_i(p)=\frac{F_i(p)}{-Q_{i,i}(p)}
 =p_i-c_i-\mu_i(p),
 \qquad
 \mu_i(p)=\frac{Q_i}{-Q_{i,i}}
 =-\frac1{\partial_{p_i}\log Q_i}.
\]
Thus $\widetilde F_i$ is actual markup minus demand-implied markup.
The multiplier $1/(-Q_{i,i})$ is positive throughout the feasible domain, so
\[
 \sgn\widetilde F_i=\sgn F_i.
\]
Since the price domain is rectangular,
\[
 \boxed{\mathrm{NE}=\Sol(K,F)=\Sol(K,\widetilde F).}
\]
This includes equilibrium prices at a floor or cap.
\end{frame}


\begin{frame}{Oligopoly: the transformed Jacobian}
\compactmath
Let $J=D\widetilde F=I-D\mu$. For every price coordinate, including $p_0$,
\[
 \frac{\partial\mu_i}{\partial p_j}
 =\frac{\partial^2\ell_i/\partial p_i\partial p_j}
 {(\partial\ell_i/\partial p_i)^2}.
\]
E1 and E2 therefore give
\[
 \boxed{
 J_{ii}=1-\partial_{p_i}\mu_i\geq1,\qquad
 J_{ij}=-\partial_{p_j}\mu_i\leq0\quad(j\ne i).}
\]
Here $J$ contains only the $n$ strategic price coordinates.
\medskip\par
The transformed Jacobian has a positive diagonal and nonpositive cross entries.
E3 controls their combined magnitude.
\end{frame}


\begin{frame}{Oligopoly: diagonal dominance and uniqueness}
\compactmath
E3 implies $\sum_{j=0}^n\partial_{p_j}\mu_i=0$. Consequently,
\[
 \boxed{
 J_{ii}-\sum_{j\ne i}|J_{ij}|
 =1+\partial_{p_0}\mu_i\geq1>0.}
\]
Positive diagonal entries and strict row diagonal dominance make $J$
a $P$-matrix at every feasible price vector.
\setcounter{proposition}{3}
\begin{proposition}[Oligopoly pricing]
Under E1--E3, the pricing game on $K=\prod_i[a_i,b_i]$, with $a_i\geq c_i$,
has \alert{at most one} pure-strategy Nash equilibrium.
\end{proposition}
Lemma 2(ii) gives uniqueness, including binding price bounds.
\end{frame}


\begin{frame}{CES complements: the original operator and Jacobian}
\small\compactmath
On a connected, nonempty undirected network without self-loops, let
\[
 u_i=a_ix_i-\tfrac12x_i^2+\delta x_iS_i(x),\qquad
 S_i(x)=\left(\sum_jg_{ij}x_j^\beta\right)^{1/\beta},
 \quad a_i,\delta,\beta>0.
\]
Every equilibrium is positive and satisfies $x_i=T_i(x):=a_i+\delta S_i(x)$.
Thus the original operator and Jacobian on $\R_{++}^n$ are
\[
 F_i(x)=x_i-a_i-\delta S_i(x),\qquad
 \boxed{DF(x)=I-\delta DS(x),}
\]
\[
 (DS(x))_{ij}=g_{ij}x_j^{\beta-1}S_i(x)^{1-\beta}.
\]
The aggregate's derivatives depend on the levels and relative sizes of efforts.
For $\beta<1$, we do not assert differentiability at zero.
\end{frame}


\begin{frame}{CES complements: SET and log coordinates}
\compactmath
First apply the increasing logarithm to the two positive terms:
\[
 \widetilde F_i(x)=\log x_i-\log T_i(x),
 \qquad
 \sgn\widetilde F_i(x)=\sgn\!\left[x_i-T_i(x)\right].
\]
Then make the one-to-one change of variables $z_i=\log x_i$:
\[
 H_i(z)=\widetilde F_i(e^z)=z_i-\log T_i(e^z),\qquad z\in\R^n.
\]
Every equilibrium has $x_i\geq a_i>0$, so
\[
 \boxed{x^*\in\mathrm{NE}\quad\Longleftrightarrow\quad H(\log x^*)=0.}
\]
The logarithmic comparison is SET. The change of coordinates is a separate step.
\end{frame}


\begin{frame}{CES complements: the new Jacobian and uniqueness}
\small\compactmath
The Jacobian in log coordinates is
\[
 \boxed{DH(z)=I-W(z),}\qquad
 W_{ij}(z)=
 \frac{\delta S_i(e^z)}{a_i+\delta S_i(e^z)}
 \frac{g_{ij}e^{\beta z_j}}{\sum_k g_{ik}e^{\beta z_k}}.
\]
Since there are no self-loops, $W_{ii}=0$. Also $W_{ij}\geq0$ and
\[
 \sum_jW_{ij}(z)=\frac{\delta S_i(e^z)}{a_i+\delta S_i(e^z)}<1.
\]
Hence, for every $z\in\R^n$,
\[
 (DH)_{ii}-\sum_{j\ne i}|(DH)_{ij}|
 =1-\sum_jW_{ij}>0.
\]
The positive diagonal and strict row dominance give a $P$-matrix.
By Lemma 2(ii), $H$ has at most one zero, so the game has \alert{at most one}
equilibrium. This is the conditional-uniqueness argument used in Theorem 4.
\end{frame}


\begin{frame}{Public goods: the original operator and Jacobian}
\small\compactmath
On an undirected network without self-loops, let
$s=(I+\delta G)x$ and $u_i(x)=b(s_i)-c(x_i)$, with $x_i\geq0$ and $0<\delta\leq1$.
\setcounter{assumption}{1}
\begin{assumption}[Benefits and costs]
$b'>0$, $b''<0$, $c'(u)>0$ for $u>0$, and $c''\geq0$, with
$b'(0)>c'(0)$ and $\lim_{u\to\infty}b'(u)<\lim_{u\to\infty}c'(u)$.
\end{assumption}
Negative marginal payoffs give
\[
 F_i(x)=c'(x_i)-b'(s_i),\qquad
 \boxed{DF(x)=C(x)+A(x)(I+\delta G),}
\]
where
\[
 C(x)=\diag(c''(x_i)),\qquad A(x)=\diag(-b''(s_i))\succ0.
\]
For $j\ne i$, $(DF)_{ij}=-\delta g_{ij}b''(s_i)$.
Its row-dependent benefit curvature generally makes $DF$ asymmetric.
\end{frame}


\begin{frame}{Public goods: SET through an inverse function}
\compactmath
Work on an equilibrium-preserving rectangle $K\subset[0,k^*]^n$ where
$c'(K_i)\subset\operatorname{range}(b')$, with $b'(k^*)=c'(k^*)$.
The inverse is finite on this domain. Define
\[
 h(u)=(b')^{-1}(c'(u)).
\]
Because $b'$ is strictly decreasing,
\[
 F_i(x)=b'(h(x_i))-b'(s_i)
 \quad\sim_{\rm sign}\quad s_i-h(x_i).
\]
The transformed operator is therefore
\[
 \boxed{\widetilde F(x)=(I+\delta G)x-h(x).}
\]
Rectangularity preserves the constrained VI solutions:
\[
 \mathrm{NE}=\Sol(K,F)=\Sol(K,\widetilde F).
\]
\end{frame}


\begin{frame}{Public goods: the transformed Jacobian}
\compactmath
Differentiate $b'(h(u))=c'(u)$:
\[
 h'(u)=\frac{c''(u)}{b''(h(u))}\leq0.
\]
The new Jacobian is
\[
 \boxed{D\widetilde F(x)=I+\delta G-\diag(h'(x_i)).}
\]
Its diagonal and cross entries are
\[
 (D\widetilde F)_{ii}=1-h'(x_i)\geq1,\qquad
 (D\widetilde F)_{ij}=\delta g_{ij}=(D\widetilde F)_{ji}\quad(j\ne i).
\]
SET removes the row-dependent benefit curvature from the cross effects.
The Jacobian is now \alert{symmetric}, making the potential route available.
\end{frame}


\begin{frame}{Public goods: positive definiteness and uniqueness}
\compactmath
Since $G=G^\top$ and $-h'(x_i)\geq0$, for every $x\in K$,
\[
 D\widetilde F(x)\succeq
 [1+\delta\lambda_{\min}(G)]I.
\]
Thus $1+\delta\lambda_{\min}(G)>0$ makes the transformed Jacobian positive definite.
Symmetry gives a potential, and positive definiteness makes it strictly convex.
\setcounter{theorem}{4}
\begin{theorem}[Public-good equilibrium]
Under Assumption 2, if $1+\delta\lambda_{\min}(G)>0$, the game has a unique
Nash equilibrium.
\end{theorem}
The Jacobian calculation supplies the uniqueness argument.
The theorem's existence argument is separate.
\end{frame}

\begin{frame}{Main conclusions}
\begin{block}{A general method}
On rectangular domains, sign-equivalent transformations preserve the entire VI
solution set and can simplify the Jacobian used to study uniqueness.
\end{block}
\begin{itemize}
 \item Oligopoly and CES complements use a transformed $P$-matrix.
 \item Public goods use symmetry and positive definiteness.
 \item The worked example has a unique equilibrium even though its original
 global $P$-matrix test fails and its best responses are not a contraction.
\end{itemize}
\medskip
The useful representation depends on the model. Existence remains a separate question.
\end{frame}

\begin{frame}{Backup: a coordinate proof of Theorem 1(ii)}
\small\compactmath
Let $K=\prod_i K_i$ and suppose $x^*\in\operatorname{Sol}(K,F)$.
For every $i$ and $t\in K_i$,
\[
 (t-x_i^*)F_i(x^*)\geq0.
\]
Sign equivalence implies the same inequality for $\widetilde F_i(x^*)$:
\[
 \begin{cases}
 F_i(x^*)=0 &\Longrightarrow\ \widetilde F_i(x^*)=0,\\
 F_i(x^*)>0 &\Longrightarrow\ t-x_i^*\geq0\text{ and }\widetilde F_i(x^*)>0,\\
 F_i(x^*)<0 &\Longrightarrow\ t-x_i^*\leq0\text{ and }\widetilde F_i(x^*)<0.
 \end{cases}
\]
Consequently, for every $x\in K$,
\[
 \langle x-x^*,\widetilde F(x^*)\rangle
 =\sum_i(x_i-x_i^*)\widetilde F_i(x^*)\geq0.
\]
Thus $\operatorname{Sol}(K,F)\subseteq\operatorname{Sol}(K,\widetilde F)$.
Interchanging the two operators gives the reverse inclusion.
\end{frame}
