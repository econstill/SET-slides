\begin{frame}{Equilibrium uniqueness and representation}
In nonlinear models, standard sufficient conditions for uniqueness
can be difficult to verify.

Some of this difficulty may come from how the equilibrium
conditions are written.

\begin{block}{The central question}
How can we transform nonlinear equilibrium to simplify analysis, without changing the equilibrium set?
\end{block}

\medskip
The paper develops \alert{sign-equivalent transformations (SET)}
to reveal useful mathematical structure in equilibrium problems.
\end{frame}


\begin{frame}{The paper's map}
\begin{center}
\includegraphics[width=0.87\textwidth]{assets/set-figure1-methodology.png}
\end{center}
\vspace{-0.25cm}
\small An equilibrium is a best-response fixed point. The figure follows its VI representation through a sign-equivalent transformation to two methods for studying uniqueness.
\end{frame}


\begin{frame}{Example 1: a two-player game}
\compactmath
Two players choose $x_i\geq0$, so $K=\mathbb R_+^2$.
\begin{block}{Payoffs}
\[
\begin{aligned}
 u_1(x)&=x_1\Bigl(1+\tfrac12x_2\Bigr)-\tfrac12x_1^2,\\
 u_2(x)&=4\Bigl[x_2\Bigl(1+\tfrac12x_1\Bigr)-\tfrac12x_2^2\Bigr].
\end{aligned}
\]
\end{block}
Each payoff is strictly concave in the player's own action.
\begin{block}{Best responses}
\[
 B_1(x_2)=1+\tfrac12x_2,\qquad
 B_2(x_1)=1+\tfrac12x_1.
\]
\end{block}
The factor $4$ changes player 2's payoff units, but leaves the best response unchanged.
\end{frame}


\begin{frame}{Example 1: the VI and its transformation}
\compactmath
For a candidate $x^*\in K$, use the negative marginal payoffs:
\[
\begin{gathered}
 F_i=-\partial_i u_i,\qquad
 F(x)=\begin{pmatrix}x_1-1-\tfrac12x_2\\4(x_2-1-\tfrac12x_1)\end{pmatrix}.\\[6pt]
 x^*\in\mathrm{NE}
 \quad\Longleftrightarrow\quad
 \langle y-x^*,F(x^*)\rangle\geq0\quad\text{for every }y\in K.
\end{gathered}
\]
\begin{block}{SET: divide the second component by $4$}
\[
 \widetilde F(x)=\operatorname{diag}(1,\tfrac14)F(x)
 =\begin{pmatrix}x_1-1-\tfrac12x_2\\x_2-1-\tfrac12x_1\end{pmatrix}.
\]
\end{block}
Positive scaling preserves every component's sign on $K$. Since $K$ is rectangular,
\[
 \operatorname{Sol}(K,F)=\operatorname{Sol}(K,\widetilde F).
\]
\end{frame}


\begin{frame}{Example 1: the two uniqueness routes}
\compactmath
The transformed system has a symmetric derivative:
\[
 H=D\widetilde F=\begin{pmatrix}1&-\tfrac12\\-\tfrac12&1\end{pmatrix},
 \qquad\lambda(H)\in\{\tfrac12,\tfrac32\}.
\]
\begin{columns}[T,onlytextwidth]
\column{0.40\textwidth}
\textbf{$P$-property method}\par
The principal minors of $H$ are $1,1,\tfrac34>0$.
\medskip\par
The $P$-matrix condition gives at most one VI solution.
\column{0.56\textwidth}
\textbf{Potential method}
\[
\begin{aligned}
 \Phi(x)={}&\tfrac12(x_1^2+x_2^2)\\
           &-\tfrac12x_1x_2-x_1-x_2.
\end{aligned}
\]
$\nabla\Phi=\widetilde F$ and $\nabla^2\Phi=H\succ0$.
The VI minimizes the strictly convex $\Phi$ over $K$.
\end{columns}
\medskip
\[
 \widetilde F(x^*)=0
 \quad\Longrightarrow\quad
 \boxed{x_1^*=x_2^*=2}.
\]
This feasible zero proves existence. The equilibrium is \alert{unique}.
\end{frame}


\begin{frame}{Example 2: different nonlinear best responses}
\compactmath
Two players choose $x_i\geq0$, with $K=\mathbb R_+^2$.
\begin{block}{Payoffs}
\[
\begin{aligned}
 u_1(x)&=x_1(1+x_2)-x_1^2-\tfrac13x_1^3,\\
 u_2(x)&=x_2\ln(2+x_1)-\tfrac12x_2^2.
\end{aligned}
\]
\end{block}
Each payoff is strictly concave in the player's own action.
\begin{block}{Best responses}
\[
 B_1(x_2)=\sqrt{2+x_2}-1,\qquad
 B_2(x_1)=\ln(2+x_1).
\]
\end{block}
Every equilibrium is interior because both best responses are positive on $K$.
\end{frame}


\begin{frame}{Example 2: a nonlinear sign-equivalent transformation}
\compactmath
The negative marginal payoffs define $F_i=-\partial_i u_i$.
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{Original VI}
\[
 F(x)=\begin{pmatrix}
 x_1^2+2x_1-1-x_2\\
 x_2-\ln(2+x_1)
 \end{pmatrix}.
\]
\[
 \partial_2F_1=-1,\quad
 \partial_1F_2=-\frac1{2+x_1}.
\]
\column{0.48\textwidth}
\textbf{After SET}
\[
 \widetilde F(x)=\begin{pmatrix}
 x_1^2+2x_1-1-x_2\\
 e^{x_2}-2-x_1
 \end{pmatrix}.
\]
\[
 \partial_2\widetilde F_1=\partial_1\widetilde F_2=-1.
\]
\end{columns}
\medskip
The exponential is strictly increasing:
\[
 x_2\gtreqless\ln(2+x_1)
 \quad\Longleftrightarrow\quad
 e^{x_2}\gtreqless2+x_1.
\]
All component signs agree on the rectangular domain $K$. Thus
\[
 \mathrm{NE}=\operatorname{Sol}(K,F)=\operatorname{Sol}(K,\widetilde F).
\]
\end{frame}


\begin{frame}{Example 2: a convex equilibrium problem}
\compactmath
The transformed operator is a gradient, $\widetilde F=\nabla\Phi$, where
\[
 \Phi(x)=\tfrac13x_1^3+x_1^2-x_1-x_1x_2+e^{x_2}-2x_2.
\]
For every $x\in K$,
\[
 \nabla^2\Phi(x)=
 \begin{pmatrix}2+2x_1&-1\\-1&e^{x_2}\end{pmatrix}
 \succeq
 \begin{pmatrix}2&-1\\-1&1\end{pmatrix}
 \succ0.
\]
\begin{block}{Unique equilibrium}
Strong convexity and coercivity give a unique minimizer on $K$.
\end{block}
SET exposes a convex optimization representation of the same equilibrium problem.
\end{frame}



\begin{frame}{Applications of general interest}
\begin{enumerate}
\item \textbf{Pricing and social interactions:} oligopoly with price bounds and nonlinear peer effects.
\item \textbf{Nonlinear network complements:} when equilibrium exists, who acts most, and who is the key player?
\item \textbf{Network public goods:} how convex costs change participation, uniqueness, and effort rankings.
\end{enumerate}
\medskip
Only one methodological fact is needed before the examples.
\end{frame}



\begin{frame}{The one methodological fact we need}
\begin{block}{Sign-equivalent transformation}
If every component of $F$ and $\widetilde F$ has the same sign at every feasible profile, then on a product of individual intervals the two variational inequalities have \alert{exactly the same solution set}.
\end{block}
\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{lll}
\toprule
Player's choice & Equilibrium condition & What matters\\
\midrule
Lower bound & $F_i\geq0$ & Sign\\
Interior & $F_i=0$ & Zero\\
Upper bound & $F_i\leq0$ & Sign\\
\bottomrule
\end{tabular}
\end{center}
The product-domain condition matters when a choice is at a boundary.
\end{frame}



\begin{frame}{What a better representation can reveal}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{$P$-property route}
A transformed derivative matrix can establish \alert{at most one} equilibrium.
\medskip
\textbf{Used for:} oligopoly pricing, peer effects, CES complements.
\end{block}
\column{0.48\textwidth}
\begin{block}{Potential route}
A symmetric transformed system can become an optimization problem; curvature can give uniqueness.
\medskip
\textbf{Used for:} network games, public goods, multiplex interactions.
\end{block}
\end{columns}
\medskip
Existence needs its own argument in each application.
\end{frame}


\begin{frame}{Oligopoly: what price should each firm charge?}
There are $n\geq2$ firms, each selling one differentiated product.
\begin{itemize}
 \item Firm $i$ has constant unit cost $c_i$ and chooses $p_i\in[a_i,b_i]$, with $a_i\geq c_i$.
 \item Demand $Q_i(p)>0$ depends on its own price and its rivals' prices.
 \item Consumers can also choose an outside option, whose price $p_0$ is fixed.
\end{itemize}
\begin{block}{The firm's problem}
\[
 \max_{p_i\in[a_i,b_i]}\;\pi_i(p)=(p_i-c_i)Q_i(p).
\]
\end{block}
Can firms' pricing decisions be mutually consistent in more than one way,
including when a price floor or cap binds?
\end{frame}


\begin{frame}{Where demand comes from}
\compactmath
Each consumer chooses the option with the highest utility:
\[
 U_i=v_i-p_i+\varepsilon_i,\qquad i=0,1,\ldots,n.
\]
Here $v_i$ measures product appeal; $\varepsilon_i$ captures individual tastes.
Option $0$ means not buying from any firm. Normalize the mass of consumers to one.
\begin{block}{A familiar example: multinomial logit}
With independent standard extreme-value taste shocks,
\[
 Q_i(p)=\frac{e^{v_i-p_i}}
 {e^{v_0-p_0}+\sum_{j=1}^n e^{v_j-p_j}}.
\]
\end{block}
A higher own price loses customers to rivals or the outside option.
The paper's argument covers more than this logit example.
\end{frame}


\begin{frame}{An optimal price balances two effects}
\compactmath
Differentiate profit, holding rivals' prices fixed:
\[
 \frac{\partial\pi_i}{\partial p_i}
 =\underbrace{Q_i}_{\text{more revenue per customer}}
 +\underbrace{(p_i-c_i)\frac{\partial Q_i}{\partial p_i}}_{\text{lost customers}},
 \qquad \frac{\partial Q_i}{\partial p_i}<0.
\]
\begin{block}{At an interior optimum}
\[
 \underbrace{p_i-c_i}_{\text{actual markup}}
 =\underbrace{\frac{Q_i}{-\partial Q_i/\partial p_i}}_{\text{demand-implied markup }\mu_i(p)}.
\]
\end{block}
$\mu_i$ is the inverse own-price \alert{semi-elasticity}, $1/[-\partial_{p_i}\log Q_i]$.
Less price-sensitive demand supports a larger markup; rivals' prices affect this sensitivity.
\end{frame}


\begin{frame}{SET: rewrite pricing as a markup gap}
\compactmath
The original VI uses negative marginal profit, $F_i=-\partial\pi_i/\partial p_i$.
Divide by the positive number $-\partial Q_i/\partial p_i$:
\begin{block}{A sign-equivalent pricing condition}
\[
 \boxed{\widetilde F_i(p)=\frac{F_i(p)}{-\partial Q_i/\partial p_i}
       =\underbrace{p_i-c_i}_{\text{actual markup}}-\underbrace{\mu_i(p)}_{\text{desired markup}}.}
\]
\end{block}
\begin{itemize}
 \item $\widetilde F_i<0$: the firm wants to \alert{raise} its price.
 \item $\widetilde F_i>0$: the firm wants to \alert{cut} its price.
 \item $\widetilde F_i=0$: the two pricing effects balance.
\end{itemize}
This is a new representation of the marginal condition, not a new profit function.
\end{frame}


\begin{frame}{Price floors and caps: signs, not just zeros}
\compactmath
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{lll}
\toprule
Chosen price & Marginal profit & Markup gap\\
\midrule
Floor: $p_i=a_i$ & $\partial\pi_i/\partial p_i\leq0$ & $\widetilde F_i\geq0$\\
Interior & $\partial\pi_i/\partial p_i=0$ & $\widetilde F_i=0$\\
Cap: $p_i=b_i$ & $\partial\pi_i/\partial p_i\geq0$ & $\widetilde F_i\leq0$\\
\bottomrule
\end{tabular}
\end{center}
Under E1 (own-price log-concavity), these conditions characterize best responses.
Positive rescaling preserves all three cases. On $K=\prod_i[a_i,b_i]$,
\[
 \mathrm{NE}=\Sol(K,F)=\Sol(K,\widetilde F).
\]
The \alert{entire equilibrium set} is preserved.
Looking only for zeros would miss equilibria with binding bounds.
\end{frame}


\begin{frame}{The demand restrictions behind uniqueness}
\compactmath
Write $\ell_i=\log Q_i$. Demand is positive, smooth, and decreasing in own price.
\begin{enumerate}
 \item \textbf{E1: own-price log-concavity}
 \[
 \frac{\partial^2\ell_i}{\partial p_i^2}\leq0.
 \]
 \item \textbf{E2: cross-price log-supermodularity}
 \[
 \frac{\partial^2\ell_i}{\partial p_i\partial p_j}\geq0
 \quad(j\ne i,\ \text{including the outside option }j=0).
 \]
 \item \textbf{E3: common-price-shift invariance}
 \[
 Q_i(p_1+t,\ldots,p_n+t,p_0+t)=Q_i(p_1,\ldots,p_n,p_0).
 \]
\end{enumerate}
The logit example satisfies all three. The outside price is fixed during the game;
E3 describes demand when \emph{all} option prices shift together.
\end{frame}


\begin{frame}{Why the transformed pricing system is unique}
\compactmath
Let $J=D\widetilde F$ be the derivative matrix with respect to firms' prices.
The demand restrictions imply
\[
 J_{ii}\geq1,\qquad J_{ij}\leq0\ (j\ne i),\qquad
 J_{ii}-\sum_{j\ne i}|J_{ij}|\geq1.
\]
An own-price change affects the markup gap more than all rival-price effects combined.
\begin{block}{The $P$-property route}
Positive diagonal and strict row diagonal dominance give the $P$-structure.
The transformed VI, and hence the pricing game, has \alert{at most one equilibrium}.
\end{block}
For finite price intervals, continuity and own-price quasi-concavity also give
existence: the equilibrium is then \alert{unique}.
\end{frame}


\begin{frame}{A two-firm logit example: solve for prices}
\compactmath
Take two identical firms: $c_1=c_2=v_1=v_2=0$ and $v_0-p_0=0$.
\[
 Q_i(p)=\frac{e^{-p_i}}{1+e^{-p_1}+e^{-p_2}},\qquad
 -\frac{\partial Q_i}{\partial p_i}=Q_i(1-Q_i).
\]
\begin{block}{The markup gap becomes especially simple}
\[
 \mu_i=\frac1{1-Q_i},\qquad
 \widetilde F_i=p_i-\frac1{1-Q_i}.
\]
\end{block}
With $p_i\in[0,4]$, look for a symmetric interior price $p$:
\[
 p=\frac{1+2e^{-p}}{1+e^{-p}}
 \quad\Longrightarrow\quad
 \boxed{p_1^*=p_2^*\approx1.227,\quad Q_1^*=Q_2^*\approx0.185.}
\]
The uniqueness result also rules out asymmetric equilibria.
\end{frame}


\begin{frame}{The same market with price floors or caps}
Keep demand and costs unchanged; change only the feasible price intervals.
\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{lcc}
\toprule
Interval for both firms & Equilibrium prices & Markup gap\\
\midrule
$[0,4]$: no binding bound & $(1.227,1.227)$ & $0$\\
$[0,1]$: cap at $1$ & $(1,1)$ & $-0.269$\\
$[1.5,4]$: floor at $1.5$ & $(1.5,1.5)$ & $+0.318$\\
\bottomrule
\end{tabular}
\end{center}
At the cap firms want higher prices; at the floor they want lower prices.
\begin{block}{What SET buys us}
Bounds change prices, but sign preservation lets the
\alert{same uniqueness argument} cover all three cases.
\end{block}
\end{frame}


\begin{frame}{Existing application: nonlinear peer effects}
\begin{block}{A social norm need not be the average peer}
\[
\widetilde x_{-i}(\beta)=\left(\sum_j\widetilde g_{ij}x_j^{\beta}\right)^{1/\beta},
\qquad
x_i=(1-\lambda_2)a_i+(\lambda_1+\lambda_2)\widetilde x_{-i}(\beta).
\]
\end{block}
\begin{itemize}
\item $\beta=1$ recovers linear-in-means; other values change whose behavior is most influential.
\item Taking logs exposes a $P$-property: under the paper's parameter restrictions, \alert{any solution is unique}.
\item The result is conditional on existence; the transformation alone does not supply it.
\end{itemize}
\end{frame}



\begin{frame}{Existing applications: network games and multiplexity}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Network public goods with income}
An inverse Engel curve turns heterogeneous marginal conditions into a symmetric system on an undirected graph.
\medskip
A potential and a spectral condition can then give uniqueness.
\end{block}
\column{0.48\textwidth}
\begin{block}{Several network layers}
An agent chooses effort across multiple kinds of ties. A transformed system can have a potential even with cross-layer costs.
\medskip
Extra curvature conditions are needed for uniqueness.
\end{block}
\end{columns}
\medskip
The same method can reveal different mathematical structures across models.
\end{frame}


\begin{frame}{CES peer effects on an unnormalized network}
\small\compactmath
\[
 u_i(x)=a_ix_i-\frac12x_i^2+
 \delta x_i\Bigl(\sum\nolimits_jg_{ij}x_j^\beta\Bigr)^{1/\beta}.
\]
\begin{block}{Best response}
\[
 T_i(x)=a_i+\delta S_i(x),\qquad
 S_i(x)=\Bigl(\sum\nolimits_jg_{ij}x_j^\beta\Bigr)^{1/\beta}.
\]
\end{block}
\begin{itemize}
 \item $a_i>0$, $\delta>0$, $\beta>0$, and $x_i\geq0$.
 \item $G$ is an undirected adjacency matrix, without self-loops.
 \item We use a \alert{connected, nonempty} network for the Perron-limit results.
\end{itemize}
\end{frame}

\begin{frame}{What the CES parameter changes}
\small\compactmath
For positive neighbor actions,
\[
 \frac{\partial S_i}{\partial x_j}
 =g_{ij}x_j^{\beta-1}S_i(x)^{1-\beta}.
\]
For two linked neighbors $j$ and $k$,
\[
 \frac{\partial S_i/\partial x_j}{\partial S_i/\partial x_k}
 =\left(\frac{x_j}{x_k}\right)^{\beta-1}.
\]
\begin{block}{Relative peer influence}
$\beta>1$ favors high-action peers at the margin.
$0<\beta<1$ places relatively more marginal weight on low-action peers.
\end{block}
Changing $\beta$ also changes the scale of this unnormalized aggregate.
\end{frame}


\begin{frame}{CES complements: uniqueness without a closed form}
\small
Every equilibrium has $x_i>0$. Write $z_i=\log x_i$ and compare the positive quantities in the best-response equation:
\[
H_i(z)=z_i-\log\!\left(a_i+\delta S_i(e^z)\right).
\]
\begin{block}{What the transformation reveals}
\[
DH(z)=I-W(z),\qquad W(z)\geq0,\qquad
\sum_jW_{ij}(z)=\frac{\delta S_i}{a_i+\delta S_i}<1.
\]
The positive baseline $a_i$ leaves some own-action weight in every row. Hence there is \alert{at most one} finite positive equilibrium.
\end{block}
Whether a finite equilibrium exists depends on the network threshold on the next slide.
\end{frame}


\begin{frame}{The spectral existence threshold}
\begin{block}{Theorem 4}
For $\bar\delta=\lambda_{\max}(G)^{-1/\beta}$:
\begin{itemize}
 \item $0<\delta<\bar\delta$: a unique positive equilibrium exists.
 \item $\delta\uparrow\bar\delta$: all actions diverge, with convergent ratios.
 \item $\delta\geq\bar\delta$: no finite positive equilibrium exists.
\end{itemize}
\end{block}
\medskip
On a $d$-regular network with $a_i=a$,
\[
 x_i^*=\frac{a}{1-\delta d^{1/\beta}},\qquad
 \bar\delta=d^{-1/\beta}.
\]
\end{frame}

\begin{frame}{Comparative statics below the threshold}
\begin{block}{How shocks change behavior}
Within the region where a finite equilibrium exists,
\[
 a'\geq a\quad\Longrightarrow\quad x^*(a',\delta,\beta)\geq x^*(a,\delta,\beta),
\]
\[
 \delta'\geq\delta\quad\Longrightarrow\quad x^*(a,\delta',\beta)\geq x^*(a,\delta,\beta).
\]
\end{block}
\begin{itemize}
 \item A higher individual baseline or stronger peer interaction raises equilibrium effort across the network.
 \item Equilibrium payoff is $u_i(x^*)=\tfrac12(x_i^*)^2$: changes in effort also change welfare.
\end{itemize}
\end{frame}

\begin{frame}{Bonacich centrality as the linear special case}
When $\beta=1$,
\[
 x=a+\delta Gx.
\]
\begin{block}{The network multiplier}
\[
 (I-\delta G)x=a,\qquad
 x^*=(I-\delta G)^{-1}a.
\]
\end{block}
\begin{itemize}
 \item $a=\mathbf1$: a standard Katz--Bonacich centrality vector.
 \item Heterogeneous $a$: baseline-weighted centrality.
 \item In the notation $(I-\delta G)x=b$, the vector $b$ is this same baseline.
\end{itemize}
\end{frame}

\begin{frame}{Eigenvector centrality at the critical limit}
\small\compactmath
For the positive unit Perron vector $e$,
\[
 Ge=\lambda_{\max}(G)e,\qquad e\gg0.
\]
\begin{block}{Theorem 4(ii)}
\[
 \lim_{\delta\uparrow\bar\delta}\frac{x_i^*}{x_j^*}
 =\left(\frac{e_i}{e_j}\right)^{1/\beta}.
\]
\end{block}
\begin{itemize}
 \item $\beta=1$: $x^*/\lVert x^*\rVert_2\to e$.
 \item General $\beta>0$: the vector $(x^*)^\beta$ approaches the Perron direction.
 \item Baseline heterogeneity affects levels, but disappears from these limiting ratios.
\end{itemize}
\end{frame}

 \frac{C}{L}=\frac{1+tq}{1+t/q},\qquad q=2^{1/(2\beta)}.
\]
\begin{block}{As $t\uparrow1$}
The ratio converges to $q$.

\smallskip
The action levels diverge.
\end{block}
At the same fraction of the threshold, smaller $\beta$ produces a larger center advantage here.
\end{columns}
\end{frame}

\begin{frame}{The key-player intervention}
Let $G^{-i}$ denote the network after removing player $i$.
\begin{block}{Total equilibrium activity lost}
\[
 \Delta_i=\sum_j x_j^*(G)-\sum_{j\ne i}x_j^*(G^{-i}),
 \qquad i^*\in\argmax_i\Delta_i.
\]
\end{block}
\begin{itemize}
 \item Removing a player eliminates their action and changes everyone else's incentives.
 \item Highest equilibrium action need not identify the largest total effect.
 \item For general $\beta$, recompute equilibrium on each remaining network.
\end{itemize}
\end{frame}

\begin{frame}{Network public goods with nonlinear costs}
\small\compactmath
On an undirected network, player $i$ chooses $x_i\geq0$ and receives
\[
 u_i(x)=b\!\left(x_i+\delta\sum_jg_{ij}x_j\right)-c(x_i),
 \qquad 0<\delta\leq1.
\]
\begin{block}{Assumption 2: benefits, costs, and a marginal crossing}
\[
 b'>0,\qquad b''<0,\qquad c'(x)>0\ (x>0),\qquad c''\geq0,
\]
\[
 b'(0)>c'(0),\qquad
 \lim_{x\to\infty}b'(x)<\lim_{x\to\infty}c'(x).
\]
\end{block}
The positive autarky effort $k^*$ uniquely solves $b'(k^*)=c'(k^*)$.

The benchmark of Bramoull\'e and Kranton (2007) has $c(x)=x$ and $\delta=1$.
\end{frame}

\begin{frame}{Best responses, corners, and existence}
\small
Let $t_i=\delta\sum_jg_{ij}x_j$ be the public good supplied by neighbors.
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{If $x_i>0$}
The agent balances marginal benefit and cost:
\[
 b'(x_i+t_i)=c'(x_i).
\]
\end{block}
\column{0.48\textwidth}
\begin{block}{If $x_i=0$}
Neighbors may already provide enough that contributing the first unit is not worthwhile.
\end{block}
\end{columns}
\medskip
More provision by neighbors weakly reduces one's best response. Every best response is at most the autarky effort $k^*$, so a finite equilibrium exists.
\end{frame}


\begin{frame}{Substitutes: curvature can restore uniqueness}
\small
The original marginal conditions are asymmetric because each agent receives different total provision. A sign-equivalent inverse transformation gives
\[
\widetilde F_i(x)=x_i+\delta\sum_jg_{ij}x_j-h(x_i),
\qquad h=(b')^{-1}\!\circ c'.
\]
\begin{block}{Economic result}
On an undirected graph this system has a potential. The sufficient condition
\[
1+\delta\lambda_{\min}(G)>0
\]
ensures a unique equilibrium. Convex private costs add curvature and can support uniqueness beyond this simple bound.
\end{block}
The sign equivalence also retains equilibria with zero effort.
\end{frame}


\begin{frame}{Zero marginal cost at zero implies full participation}
\small\compactmath
Now suppose $c$ is strictly convex, $c'(0)=0$, and $\delta=1$,
as in Proposition 7.
\begin{block}{Every equilibrium is interior}
At any finite profile with $x_i=0$,
\[
 \left.\frac{\partial u_i}{\partial x_i}\right|_{x_i=0}
   =b'\!\Bigl(\sum\nolimits_jg_{ij}x_j\Bigr)-c'(0)>0.
\]
Hence $x_i^*>0$ for every player at every Nash equilibrium.
\end{block}
All equilibrium conditions therefore hold with equality:
\[
 b'\!\Bigl(x_i^*+\sum\nolimits_jg_{ij}x_j^*\Bigr)=c'(x_i^*).
\]
Full participation does not require equilibrium uniqueness.
\end{frame}

\begin{frame}{Complete networks: convex costs remove the continuum}
\small\compactmath
Take the complete network and $\delta=1$. Every player receives total provision
$S=\sum_jx_j$.
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Strictly convex costs, $c'(0)=0$}
\[
 c'(x_i)=b'(S)\quad\text{for all }i.
\]
Strictly increasing $c'$ forces equal effort.
\[
 x_i=\bar x,\qquad b'(n\bar x)=c'(\bar x).
\]
The equilibrium is unique.
\end{block}
\column{0.48\textwidth}
\begin{block}{Linear costs, $c(x)=x$}
With $b'(k^*)=1$, every profile satisfying
\[
 x_i\geq0,\qquad \sum_ix_i=k^*
\]
is an equilibrium.

The continuum includes both equal sharing and a single specialist.
\end{block}
\end{columns}
Here $\lambda_{\min}(G)=-1$. Uniqueness with strictly convex costs holds even
though $1+\delta\lambda_{\min}(G)=0$.
\end{frame}

\begin{frame}{Network structure determines effort rankings}
\small\compactmath
Maintain homogeneous $b,c$, strictly convex costs, $c'(0)=0$, and $\delta=1$.
Let $N_i$ and $\overline N_i=N_i\cup\{i\}$ denote open and closed neighborhoods.
\begin{block}{Neighborhood inclusion}
At every equilibrium,
\[
 N_i\subsetneq N_j\ \text{ or }\ 
 \overline N_i\subsetneq\overline N_j
 \quad\Longrightarrow\quad x_i^*>x_j^*.
\]
\end{block}
\begin{itemize}
 \item \textbf{Star, $n\geq3$.} Leaves choose equal effort. The hub contributes less.
 \item \textbf{Nested split graph.} If $d_i<d_j$, then $x_i^*>x_j^*$.
\end{itemize}
Degree alone does not imply this ranking on arbitrary networks.
\end{frame}

\begin{block}{A general method}
On rectangular domains, the signs of the equilibrium operator determine the VI solution set.
Suitable transformations can reveal structure absent from the original Jacobian.
\end{block}

\begin{frame}{Backup: the oligopoly $P$-matrix calculation}
\small\compactmath
Let $\ell_{i,j}=\partial\log Q_i/\partial p_j$ and
$\ell_{i,jk}=\partial^2\log Q_i/(\partial p_j\partial p_k)$. Then
\[
 \widetilde F_i=p_i-c_i+\frac1{\ell_{i,i}},\qquad
 J_{ij}=\mathbf1_{\{i=j\}}-\frac{\ell_{i,ij}}{(\ell_{i,i})^2}.
\]
E1 gives $J_{ii}\geq1$; E2 gives $J_{ij}\leq0$ for $j\ne i$.
Differentiate E3's common-shift identity, then differentiate with respect to $p_i$:
\[
 \sum_{j=0}^n\ell_{i,j}=0
 \quad\Longrightarrow\quad
 \sum_{j=0}^n\ell_{i,ij}=0.
\]
Since E2 also gives $\ell_{i,i0}\geq0$,
\[
 J_{ii}-\sum_{j\ne i}|J_{ij}|
 =\sum_{j=1}^nJ_{ij}
 =1+\frac{\ell_{i,i0}}{(\ell_{i,i})^2}\geq1.
\]
Every principal submatrix has positive diagonal and strict row diagonal dominance,
so every principal determinant is positive. Thus $J$ is a $P$-matrix.
The uniform dominance also gives the $P$-property of $\widetilde F$ on the price box.
\end{frame}

\begin{frame}{Backup: logit checks and the role of SET}
\small\compactmath
For unit-slope logit, let $Q_0$ denote the outside-option share. Then
\[
 \ell_{i,i}=-(1-Q_i),\qquad
 \ell_{i,ii}=-Q_i(1-Q_i)\leq0,\qquad
 \ell_{i,ij}=Q_iQ_j\geq0\quad(j\ne i,\ j=0\text{ allowed}).
\]
Adding $t$ to every option's price multiplies every logit weight by $e^{-t}$,
so shares are unchanged. Thus E1--E3 hold.
\begin{block}{The original derivative need not be a $P$-matrix}
In the two-firm example, take $p_1=p_2=4$ and write $s=Q_1=Q_2$.
For $F_i=-\partial\pi_i/\partial p_i$,
\[
 \frac{\partial F_i}{\partial p_i}
 =s(1-s)\bigl[2-p_i(1-2s)\bigr]<0.
\]
A negative $1\times1$ principal minor rules out a global $P$-matrix test for $DF$.
The transformed derivative passes the test throughout the same price box.
\end{block}
E1 makes profit log-concave for $p_i>c_i$, not necessarily concave:
$\log\pi_i=\log(p_i-c_i)+\log Q_i$.
\end{frame}

\begin{frame}{Backup: a formal coordinate proof of Theorem 1}
\small\compactmath
Let $K=\prod_i K_i$ and suppose $x^*\in\operatorname{Sol}(K,F)$.
For every $i$ and $t\in K_i$,
\[
 (t-x_i^*)F_i(x^*)\geq0.
\]
Sign equivalence implies the same inequality for $\widetilde F_i(x^*)$:
\[
 \begin{cases}
 F_i(x^*)=0 &\Longrightarrow\ \widetilde F_i(x^*)=0,\\
 F_i(x^*)>0 &\Longrightarrow\ t-x_i^*\geq0\text{ and }\widetilde F_i(x^*)>0,\\
 F_i(x^*)<0 &\Longrightarrow\ t-x_i^*\leq0\text{ and }\widetilde F_i(x^*)<0.
 \end{cases}
\]
Consequently, for every $x\in K$,
\[
 \langle x-x^*,\widetilde F(x^*)\rangle
 =\sum_i(x_i-x_i^*)\widetilde F_i(x^*)\geq0.
\]
Thus $\operatorname{Sol}(K,F)\subseteq\operatorname{Sol}(K,\widetilde F)$.
Interchanging the two operators gives the reverse inclusion.
\end{frame}

\begin{frame}{Backup: matrix tests in the payoff-scaling example}
\small\compactmath
For $J=\begin{pmatrix}1&-\delta\\-\kappa\delta&\kappa\end{pmatrix}$ with $\kappa>0$:
\begin{columns}[T,onlytextwidth]
\column{0.47\textwidth}
\begin{block}{P-matrix test}
The principal minors are
\[
 1,\quad\kappa,\quad\kappa(1-\delta^2).
\]
All are positive if $0<\delta<1$.
\end{block}
\column{0.47\textwidth}
\begin{block}{Strict monotonicity test}
For $S=(J+J^\top)/2$,
\[
 \det S=\kappa-\frac{(1+\kappa)^2\delta^2}{4}.
\]
$S\succ0$ iff $\delta<2\sqrt\kappa/(1+\kappa)$.
\end{block}
\end{columns}
Let $\Delta=(1-\kappa)^2+(1+\kappa)^2\delta^2$. The smallest eigenvalue is
\[
 \lambda_{\min}(S)=\frac{1+\kappa-\sqrt{\Delta}}{2}.
\]
Rescaling by $\operatorname{diag}(1,1/\kappa)$ restores the symmetric derivative
with eigenvalues $1-\delta$ and $1+\delta$.
\end{frame}

\begin{frame}{Backup: the nonlinear Perron limit}
\small\compactmath
Let $y(\delta)=x(\delta)^\beta$ and normalize $q(\delta)=y(\delta)/\lVert y(\delta)\rVert_1$.
\begin{itemize}
 \item If actions remained bounded as $\delta\uparrow\bar\delta$, continuity would give an equilibrium at $\bar\delta$.
 \item Connectedness propagates divergence: $x_i\geq\delta x_j$ whenever $g_{ij}=1$.
\end{itemize}
Therefore $a_i/x_i\to0$, and along any convergent subsequence,
\[
 \frac{(x-a)^\beta}{\lVert x^\beta\rVert_1}
 =\delta^\beta Gq(\delta)
 \quad\Longrightarrow\quad q=\bar\delta^\beta Gq.
\]
\begin{block}{Uniqueness of the normalized limit}
The only nonnegative unit-$\ell_1$ Perron vector is $e/\lVert e\rVert_1$.
Every subsequential limit is the same, so the full sequence converges.
\end{block}
\end{frame}

\begin{frame}{Backup: complete-network uniqueness without the strict spectral bound}
\small\compactmath
Assume homogeneous $b,c$, strictly convex $c$, $c'(0)=0$, and $\delta=1$.
On the complete network, write $S=\sum_jx_j$.
\begin{enumerate}
 \item Full participation makes every first-order condition an equality:
 \[
 c'(x_i)=b'(S)=c'(x_j)\qquad\text{for every }i,j.
 \]
 \item Strictly increasing $c'$ implies $x_i=x_j=\bar x$.
 \item The function $f(z)=b'(nz)-c'(z)$ is strictly decreasing, with
 $f(0)>0$ and an eventually negative value. Thus it has exactly one positive zero.
\end{enumerate}
\begin{block}{Why the spectral test is only sufficient}
For $n\geq2$, the complete graph has $\lambda_{\min}(G)=-1$.
At $\delta=1$ the strict spectral inequality fails, but the argument above proves uniqueness.
\end{block}
\end{frame}

\begin{frame}{Backup: why the hub contributes less in a star}
\small\compactmath
Let $n\geq3$ and $\delta=1$, with homogeneous functions, strictly convex costs,
and $c'(0)=0$. Write $x_H$ for the hub's effort.
\begin{enumerate}
 \item Every leaf faces the same peer effort $x_H$ and has a unique best response.
 Thus all leaves choose one common effort $x_L>0$.
 \item The hub and a leaf satisfy
 \[
 c'(x_H)=b'\!\left(x_H+(n-1)x_L\right),\qquad
 c'(x_L)=b'(x_H+x_L).
 \]
 \item Since $(n-2)x_L>0$ and $b'$ is strictly decreasing,
 \[
 c'(x_H)<c'(x_L).
 \]
 Strictly increasing $c'$ then gives $\boxed{x_H<x_L}$.
\end{enumerate}
The proof applies to every equilibrium, without a uniqueness assumption.
\end{frame}
